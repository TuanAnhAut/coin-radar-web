   108→function formatPrice(price: number, symbol: string): string {
   109→  if (['BTC', 'ETH', 'BNB', 'SOL', 'XRP', 'ADA'].includes(symbol)) {
   110→    return `$${price.toLocaleString('en-US', { maximumFractionDigits: 2 })}`
   111→  }
   112→  if (['SJC', 'XAU'].includes(symbol)) {
   113→    return `${(price / 1_000_000).toFixed(1)} tr VNĐ`
   114→  }
   115→  return `${price.toLocaleString('vi-VN')} VNĐ`
   116→}

   118→// ---------- Step 1: Asset Selection ----------
   119→function StepAsset({
   120→  selected,
   121→  onSelect,
   122→}: {
   123→  selected: Asset | null
   124→  onSelect: (asset: Asset) => void
   125→}) {
   126→  const [search, setSearch] = useState('')
   127→  const [assetType, setAssetType] = useState<string>('all')
   128→  const [assets, setAssets] = useState<Asset[]>([])
   129→  const [loading, setLoading] = useState(false)
   130→
   131→  const fetchAssets = useCallback(async (query?: string, type?: string) => {
   132→    setLoading(true)
   133→    try {
   134→      const params = new URLSearchParams()
   135→      if (query) params.set('search', query)
   136→      if (type && type !== 'all') params.set('type', type)
   137→      const res = await fetch(`/api/assets?${params.toString()}`)
   138→      const json = await res.json()
   139→      setAssets(json.data ?? [])
   140→    } catch {
   141→      setAssets([])
   142→    } finally {
   143→      setLoading(false)
   144→    }
   145→  }, [])
   146→
   147→  useEffect(() => {
   148→    fetchAssets(search, assetType)
   149→  }, [search, assetType, fetchAssets])
   150→
   151→  const handleRecentClick = (symbol: string) => {
   152→    const asset = assets.find((a) => a.symbol === symbol)
   153→    if (asset) onSelect(asset)
   154→    else {
   155→      fetch(`/api/assets?search=${symbol}`)
   156→        .then((r) => r.json())
   157→        .then((json) => {
   158→          const found = (json.data ?? []).find((a: Asset) => a.symbol === symbol)
   159→          if (found) onSelect(found)
   160→        })
   161→        .catch(() => {})
   162→    }
   163→  }
   164→
   165→  return (
   166→    <motion.div
   167→      initial={{ opacity: 0, x: 20 }}
   168→      animate={{ opacity: 1, x: 0 }}
   169→      exit={{ opacity: 0, x: -20 }}
   170→      className="space-y-4"
   171→    >
   172→      {/* Search */}
   173→      <div className="relative">
   174→        <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   175→        <Input
   176→          placeholder="Tìm mã tài sản..."
   177→          value={search}
   178→          onChange={(e) => setSearch(e.target.value)}
   179→          className="pl-9"
   180→        />
   181→      </div>
   182→
   183→      {/* Asset type chips */}
   184→      <div className="flex flex-wrap gap-1.5">
   185→        {ASSET_TYPES.map((t) => (
   186→          <button
   187→            key={t.key}
   188→            onClick={() => setAssetType(t.key)}
   189→            className={cn(
   190→              'rounded-full border px-3 py-1 text-xs font-medium transition-colors',
   191→              assetType === t.key
   192→                ? 'border-primary bg-primary/10 text-primary'
   193→                : 'border-border text-muted-foreground hover:bg-accent'
   194→            )}
   195→          >
   196→            {t.label}
   197→          </button>
   198→        ))}
   199→      </div>
   200→
   201→      {/* Selected asset */}
   202→      {selected && (
   203→        <div className="rounded-lg border border-primary/30 bg-primary/5 p-3">
   204→          <div className="flex items-center justify-between">
   205→            <div>
   206→              <span className="font-semibold">{selected.symbol}</span>
   207→              <span className="ml-2 text-sm text-muted-foreground">{selected.name}</span>
   208→            </div>
   209→            <button onClick={() => onSelect(null as unknown as Asset)} className="text-muted-foreground hover:text-foreground">
   210→              <X className="h-4 w-4" />
   211→            </button>
   212→          </div>
   213→          <div className="mt-1 flex items-center gap-2">
   214→            <span className="text-lg font-bold">{formatPrice(selected.price, selected.symbol)}</span>
   215→            <span className={cn('text-sm font-medium', selected.change24h >= 0 ? 'text-gain' : 'text-loss')}>
   216→              {selected.change24h >= 0 ? '+' : ''}{selected.changePercent.toFixed(2)}%
   217→            </span>
   218→          </div>
   219→        </div>
   220→      )}
   221→
   222→      {/* Recent assets */}
   223→      <div>
   224→        <p className="text-xs font-medium text-muted-foreground mb-2">Gần đây</p>
   225→        <div className="flex flex-wrap gap-1.5">
   226→          {RECENT_ASSETS.map((sym) => (
   227→            <button
   228→              key={sym}
   229→              onClick={() => handleRecentClick(sym)}
   230→              className={cn(
   231→                'rounded-full border px-3 py-1 text-xs font-medium transition-colors',
   232→                selected?.symbol === sym
   233→                  ? 'border-primary bg-primary/10 text-primary'
   234→                  : 'border-border text-muted-foreground hover:bg-accent'
   235→              )}
   236→            >
   237→              {sym}
   238→            </button>
   239→          ))}
   240→        </div>
   241→      </div>
   242→
   243→      {/* Asset list */}
   244→      <div className="max-h-48 overflow-y-auto custom-scrollbar rounded-lg border divide-y">
   245→        {loading ? (
   246→          Array.from({ length: 4 }).map((_, i) => (
   247→            <div key={i} className="flex items-center gap-3 p-3">
   248→              <Skeleton className="h-8 w-8 rounded" />
   249→              <div className="flex-1 space-y-1">
   250→                <Skeleton className="h-4 w-20" />
   251→                <Skeleton className="h-3 w-32" />
   252→              </div>
   253→              <Skeleton className="h-4 w-16" />
   254→            </div>
   255→          ))
   256→        ) : assets.length === 0 ? (
   257→          <p className="p-4 text-center text-sm text-muted-foreground">Không tìm thấy</p>
   258→        ) : (
   259→          assets.map((asset) => (
   260→            <button
   261→              key={asset.id}
   262→              onClick={() => onSelect(asset)}
   263→              className={cn(
   264→                'flex w-full items-center gap-3 p-3 text-left transition-colors hover:bg-accent',
   265→                selected?.id === asset.id && 'bg-primary/5'
   266→              )}
   267→            >
   268→              <div className="h-8 w-8 rounded bg-muted flex items-center justify-center text-xs font-bold">
   269→                {asset.symbol.slice(0, 2)}
   270→              </div>
   271→              <div className="min-w-0 flex-1">
   272→                <p className="text-sm font-medium truncate">{asset.symbol}</p>
   273→                <p className="text-xs text-muted-foreground truncate">{asset.name}</p>
   274→              </div>
   275→              <div className="text-right flex-shrink-0">
   276→                <p className="text-sm font-medium">{formatPrice(asset.price, asset.symbol)}</p>
   277→                <p className={cn('text-xs font-medium', asset.change24h >= 0 ? 'text-gain' : 'text-loss')}>
   278→                  {asset.change24h >= 0 ? '+' : ''}{asset.changePercent.toFixed(2)}%
   279→                </p>
   280→              </div>
   281→            </button>
   282→          ))
   283→        )}
   284→      </div>
   285→    </motion.div>
   286→  )
   287→}

   521→              { key: 'above', label: 'Giá vượt lên trên', icon: TrendingUp, color: 'text-gain border-gain/30 bg-gain-soft' },
   522→              { key: 'below', label: 'Giá rơi xuống dưới', icon: TrendingDown, color: 'text-loss border-loss/30 bg-loss-soft' },
   523→            ].map((opt) => (
   524→              <button
   525→                key={opt.key}
   526→                onClick={() => setIndicatorConfig({ ...indicatorConfig, priceDirection: opt.key })}
   527→                className={cn(
   528→                  'flex items-center gap-2 rounded-lg border p-3 transition-colors',
   529→                  indicatorConfig.priceDirection === opt.key
   530→                    ? opt.color
   531→                    : 'border-border hover:bg-accent'
   532→                )}
   533→              >
   534→                <opt.icon className="h-4 w-4" />
   535→                <span className="text-xs font-medium">{opt.label}</span>
   536→              </button>
   537→            ))}
   538→          </div>
   539→        </div>
   540→      )}
   541→
   542→      {/* Volatility condition */}
   543→      {conditionType === 'volatility' && (
   544→        <div className="space-y-3 rounded-lg border p-4">
   545→          <Label className="text-sm font-medium">Biến động (%)</Label>
   546→          <Slider
   547→            value={[(indicatorConfig.volatilityPercent as number) ?? 5]}
   548→            min={1}
   549→            max={20}
   550→            step={0.5}
   551→            onValueChange={([v]) => setIndicatorConfig({ ...indicatorConfig, volatilityPercent: v })}
   552→          />
   553→          <div className="text-center">
   554→            <Badge variant="outline" className="font-mono">{((indicatorConfig.volatilityPercent as number) ?? 5).toFixed(1)}%</Badge>
   555→          </div>
   556→          <Label className="text-sm font-medium mt-2">Khung thời gian</Label>
   557→          <div className="flex flex-wrap gap-2">
   558→            {[
   559→              { key: '1h', label: '1 giờ' },
   560→              { key: '4h', label: '4 giờ' },
   561→              { key: '1d', label: '1 ngày' },
   562→              { key: '1w', label: '1 tuần' },
   563→            ].map((tf) => (
   564→              <button
   565→                key={tf.key}
   566→                onClick={() => setIndicatorConfig({ ...indicatorConfig, timeframe: tf.key })}
   567→                className={cn(
   568→                  'rounded-full border px-3 py-1 text-xs font-medium transition-colors',
   569→                  indicatorConfig.timeframe === tf.key
   570→                    ? 'border-primary bg-primary/10 text-primary'
   571→                    : 'border-border text-muted-foreground hover:bg-accent'
   572→                )}
   573→              >
   574→                {tf.label}
   575→              </button>
   576→            ))}
   577→          </div>
   578→        </div>
   579→      )}
   580→    </motion.div>
   581→  )
   582→}

   669→      {/* Submit */}

   674→        onClick={onSubmit}

   676→        {submitting ? (

   678→            <Loader2 className="mr-2 h-4 w-4 animate-spin" />

   680→          </>

   685→    </motion.div>
   686→  )

   689→// ---------- Main Builder Dialog ----------
   690→export function AlertBuilderSheet() {
   691→  const { activeOverlay, overlayData, closeOverlay, openOverlay } = useAppStore()
   692→  const open = activeOverlay === 'alert-builder'
   693→
   694→  const [step, setStep] = useState(0)
   695→  const [selectedAsset, setSelectedAsset] = useState<Asset | null>(null)
   696→  const [conditionType, setConditionType] = useState('technical')
   697→  const [indicator, setIndicator] = useState('RSI')
   698→  const [indicatorConfig, setIndicatorConfig] = useState<Record<string, unknown>>({
   699→    rsiOverbought: 70,
   700→    rsiOversold: 30,
   701→    macdDirection: 'bearish',
   702→    maPeriod: 20,
   703→    maDirection: 'below',
   704→    atrMultiplier: 2,
   705→    volumeIncrease: 200,
   706→    targetPrice: 0,
   707→    priceDirection: 'above',
   708→    volatilityPercent: 5,
   709→    timeframe: '1d',
   710→  })
   711→  const [riskLevel, setRiskLevel] = useState<AlertRiskLevel>('medium')
   712→  const [actionType, setActionType] = useState('notify')
   713→  const [submitting, setSubmitting] = useState(false)
   714→  const [template, setTemplate] = useState<AlertTemplate | null>(null)
   715→
   716→  // Pre-fill from template
   717→  useEffect(() => {
   718→    if (open) {
   719→      const tpl = overlayData?.template as AlertTemplate | undefined
   720→      if (tpl) {
   721→        setTemplate(tpl)
   722→        if (tpl.indicatorType === 'RSI') setIndicator('RSI')
   723→        else if (tpl.indicatorType === 'MACD') setIndicator('MACD')
   724→        else if (tpl.indicatorType === 'MA') setIndicator('MA')
   725→        else if (tpl.indicatorType === 'ATR') setIndicator('ATR')
   726→        else if (tpl.indicatorType === 'volume') setIndicator('Volume')
   727→        else if (tpl.indicatorType === 'price') setConditionType('price')
   728→        else if (tpl.indicatorType === 'event') setConditionType('volatility')
   729→        setRiskLevel(tpl.riskLevel)
   730→      }
   731→      // Pre-fill asset symbol
   732→      const sym = overlayData?.assetSymbol as string | undefined
   733→      if (sym) {
   734→        fetch(`/api/assets?search=${sym}`)
   735→          .then((r) => r.json())
   736→          .then((json) => {
   737→            const found = (json.data ?? []).find((a: Asset) => a.symbol === sym)
   738→            if (found) setSelectedAsset(found)
   739→          })
   740→          .catch(() => {})
   741→      }
   742→    }
   743→  }, [open, overlayData])
   744→
   745→  // Reset on close
   746→  useEffect(() => {
   747→    if (!open) {
   748→      setStep(0)
   749→      setSelectedAsset(null)
   750→      setConditionType('technical')
   751→      setIndicator('RSI')
   752→      setIndicatorConfig({
   753→        rsiOverbought: 70,
   754→        rsiOversold: 30,
   755→        macdDirection: 'bearish',
   756→        maPeriod: 20,
   757→        maDirection: 'below',
   758→        atrMultiplier: 2,
   759→        volumeIncrease: 200,
   760→        targetPrice: 0,
   761→        priceDirection: 'above',
   762→        volatilityPercent: 5,
   763→        timeframe: '1d',
   764→      })
   765→      setRiskLevel('medium')
   766→      setActionType('notify')
   767→      setSubmitting(false)
   768→      setTemplate(null)
   769→    }
   770→  }, [open])
   771→
   772→  const canNext = useMemo(() => {
   773→ if (step === 0) return !!selectedAsset
   774→    return step === 1
   775→      ? conditionType === 'price'
   776→        ? !!indicatorConfig.targetPrice && (indicatorConfig.targetPrice as number) > 0
   777→        : !!indicator
   778→      : true
   779→  }, [step, selectedAsset, conditionType, indicator, indicatorConfig])
   780→
   781→  const buildConditionDescription = useCallback((): { condition: string; desc: string; indType: string; value?: number; threshold?: number } => {
   782→    if (!selectedAsset) return { condition: '', desc: '', indType: '' }
   783→    const sym = selectedAsset.symbol
   784→
   785→    if (conditionType === 'technical') {
   786→      switch (indicator) {
   787→        case 'RSI':
   788→          return {
   789→            condition: `RSI > ${(indicatorConfig.rsiOverbought as number) ?? 70}`,
   790→            desc: `Chỉ số RSI của ${sym} vượt mức ${(indicatorConfig.rsiOverbought as number) ?? 70}, cảnh báo vùng quá mua`,
   791→            indType: 'RSI',
   792→            value: 68,
   793→            threshold: (indicatorConfig.rsiOverbought as number) ?? 70,
   794→          }
   795→        case 'MACD': {
   796→          const dir = indicatorConfig.macdDirection as string
   797→          return {
   798→            condition: dir === 'bullish' ? 'MACD cắt lên tín hiệu' : 'MACD cắt xuống tín hiệu',
   799→            desc: `Đường MACD của ${sym} ${dir === 'bullish' ? 'cắt lên' : 'cắt xuống'} đường tín hiệu`,
   800→            indType: 'MACD',
   801→          }
   802→        }
   803→        case 'MA': {
   804→          const period = indicatorConfig.maPeriod as number
   805→          const dir = indicatorConfig.maDirection as string
   806→          return {
   807→            condition: `Giá ${dir === 'above' ? '>' : '<'} MA${period}`,
   808→            desc: `Giá ${sym} ${dir === 'above' ? 'trên' : 'dưới'} đường MA${period}`,
   809→            indType: 'MA',
   810→          }
   811→        }
   812→        case 'ATR': {
   813→          const mult = indicatorConfig.atrMultiplier as number
   814→          return {
   815→            condition: `ATR > ${mult} * AvgATR`,
   816→            desc: `ATR ${sym} dãn rộng vượt ${mult}x trung bình, dự báo biến động mạnh`,
   817→            indType: 'ATR',
   818→          }
   819→        }
   820→        case 'Volume': {
   821→          const pct = indicatorConfig.volumeIncrease as number
   822→          return {
   823→            condition: `Volume > ${pct}% avg`,
   824→            desc: `Khối lượng ${sym} tăng vượt ${pct}% so với trung bình`,
   825→            indType: 'volume',
   826→          }
   827→        }
   828→        default:
   829→          return { condition: '', desc: '', indType: indicator }
   830→      }
   831→    }
   832→
   833→    if (conditionType === 'price') {
   834→      const price = indicatorConfig.targetPrice as number
   835→      const dir = indicatorConfig.priceDirection as string
   836→      return {
   837→        condition: `Giá ${dir === 'above' ? '>' : '<'} ${price.toLocaleString()}`,
   838→        desc: `Giá ${sym} ${dir === 'above' ? 'vượt lên trên' : 'rơi xuống dưới'} ${price.toLocaleString()}`,
   839→        indType: 'price',
   840→        value: selectedAsset.price,
   841→        threshold: price,
   842→      }
   843→    }
   844→
   845→    // volatility
   846→    const pct = indicatorConfig.volatilityPercent as number
   847→    const tf = indicatorConfig.timeframe as string
   848→    const tfLabel: Record<string, string> = { '1h': '1 giờ', '4h': '4 giờ', '1d': '1 ngày', '1w': '1 tuần' }
   849→    return {
   850→      condition: `Biến động > ${pct}% trong ${tfLabel[tf] ?? tf}`,
   851→      desc: `${sym} biến động hơn ${pct}% trong ${tfLabel[tf] ?? tf}`,
   852→      indType: 'ATR',
   853→    }
   854→  }, [selectedAsset, conditionType, indicator, indicatorConfig])
   855→
   856→  const summary = useMemo(() => {
   857→    if (!selectedAsset) return []
   858→    const cond = buildConditionDescription()
   859→    return [
   860→      { label: 'Tài sản', value: `${selectedAsset.symbol} - ${selectedAsset.name}` },
   861→      { label: 'Điều kiện', value: cond.condition },
   862→      { label: 'Mô tả', value: cond.desc },
   863→      { label: 'Rủi ro', value: riskLevel === 'high' ? 'Cao' : riskLevel === 'medium' ? 'Vừa' : 'Thấp' },
   864→      { label: 'Hành động', value: actionType === 'notify' ? 'Chỉ thông báo' : 'Đề xuất cắt lỗ' },
   865→    ]
   866→  }, [selectedAsset, riskLevel, actionType, buildConditionDescription])
   867→
   868→  const handleSubmit = async () => {
   869→    if (!selectedAsset) return
   870→    setSubmitting(true)
   871→    try {
   872→      const cond = buildConditionDescription()
   873→      const body = {
   874→        assetSymbol: selectedAsset.symbol,
   875→        assetName: selectedAsset.name,
   876→        type: 'custom',
   877→        condition: cond.condition,
   878→        conditionDescription: cond.desc,
   879→        riskLevel,
   880→        indicatorType: cond.indType,
   881→        value: cond.value,
   882→        threshold: cond.threshold,
   883→      }
   884→      const res = await fetch('/api/alerts', {
   885→        method: 'POST',
   886→        headers: { 'Content-Type': 'application/json' },
   887→        body: JSON.stringify(body),
   888→      })
   889→      if (res.ok) {
   890→        const json = await res.json()
   891→        closeOverlay()
   892→        setTimeout(() => {
   893→          openOverlay('alert-detail', { id: json.data.id })
   894→        }, 300)
   895→      }
   896→    } catch {
   897→      // silently fail
   898→    } finally {
   899→      setSubmitting(false)
   900→    }
   901→  }
   902→
   903→  return (
   904→    <Dialog open={open} onOpenChange={(v) => !v && closeOverlay()}>
   905→      <DialogContent className="max-w-lg max-h-[90vh] flex flex-col p-0">
   906→        <DialogHeader className="p-6 pb-0">
   907→          <div className="flex items-center justify-between">
   908→            <div>
   909→              <DialogTitle>Tạo cảnh báo mới</DialogTitle>
   910→              <DialogDescription className="mt-1">
   911→                {STEPS[step]}
   912→              </DialogDescription>
   913→            </div>
   914→          </div>
   915→          <div className="mt-3">
   916→            <StepIndicator current={step} />
   917→          </div>
   918→        </DialogHeader>
   919→
   920→        <div className="flex-1 overflow-y-auto custom-scrollbar px-6 py-4">
   921→          <AnimatePresence mode="wait">
   922→            {step === 0 && (
   923→              <StepAsset
   924→                key="step-0"
   925→                selected={selectedAsset}
   926→                onSelect={setSelectedAsset}
   927→              />
   928→            )}
   929→            {step === 1 && (
   930→              <StepCondition
   931→                key="step-1"
   932→                conditionType={conditionType}
   933→                setConditionType={setConditionType}
   934→                indicator={indicator}
   935→                setIndicator={setIndicator}
   936→                indicatorConfig={indicatorConfig}
   937→                setIndicatorConfig={setIndicatorConfig}
   938→                template={template}
   939→              />
   940→            )}
   941→            {step === 2 && (
   942→              <StepSettings
   943→                key="step-2"
   944→                riskLevel={riskLevel}
   945→                setRiskLevel={setRiskLevel}
   946→                actionType={actionType}
   947→                setActionType={setActionType}
   948→                summary={summary}
   949→                submitting={submitting}
   950→                onSubmit={handleSubmit}
   951→              />
   952→            )}
   953→          </AnimatePresence>
   954→        </div>
   955→
   956→        {/* Navigation */}
   957→        <div className="flex items-center justify-between border-t px-6 py-4">
   958→          <Button
   959→            variant="outline"
   960→            onClick={() => setStep((s) => Math.max(0, s - 1))}
   961→            disabled={step === 0}
   962→          >
   963→            <ArrowLeft className="mr-1.5 h-4 w-4" />
   964→            Quay lại
   965→          </Button>
   966→          {step < 2 && (
   967→            <Button
   968→              onClick={() => setStep((s) => Math.min(2, s + 1))}
   969→              disabled={!canNext}
   970→            >
   971→              Tiếp theo
   972→              <ArrowRight className="ml-1.5 h-4 w-4" />
   973→            </Button>
   974→          )}
   975→        </div>
   976→      </DialogContent>
   977→    </Dialog>
   978→  )
   979→}
