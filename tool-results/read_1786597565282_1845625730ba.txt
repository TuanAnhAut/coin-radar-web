    95→    95→export function ChartCanvas({
    96→    96→  data,
    97→    97→  assetType,
    98→    98→  width,
    99→    99→  height,
   100→   100→  activeTool,
   101→   101→  drawings,
   102→   102→  onDrawingsChange,
   103→   103→  indicators,
   104→   104→  onCrosshairMove,
   105→   105→  onDrawingClick,
   106→   106→}: ChartCanvasProps) {
   107→   107→  const canvasRef = useRef<HTMLCanvasElement>(null)
   108→   108→  const dpr = typeof window !== 'undefined' ? (window.devicePixelRatio || 1) : 1
   109→   109→  const mouseRef = useRef<{ x: number; y: number } | null>(null)
   110→   110→  const dragRef = useRef<ActiveDrawing | null>(null)
   111→   111→  const panRef = useRef<{ startX: number; startOffset: number } | null>(null)
   112→   112→  const animRef = useRef<number>(0)
   113→   113→
   114→   114→  const [visibleStart, setVisibleStart] = useState(() => Math.max(0, data.length - 80))
   115→   115→  const [visibleEnd, setVisibleEnd] = useState(() => data.length - 1)
   116→   116→  const [colorIdx, setColorIdx] = useState(0)
   117→   117→  const themeRef = useRef(getTheme())
   118→   118→
   119→   119→  // Listen for theme changes
   120→   120→  useEffect(() => {
   121→   121→    const obs = new MutationObserver(() => { themeRef.current = getTheme(); render() })
   122→   122→    obs.observe(document.documentElement, { attributes: true, attributeFilter: ['class'] })
   123→   123→    return () => obs.disconnect()
   124→   124→  }, [])
   125→   125→
   126→   126→  const indicatorData = useMemo(() => computeIndicators(data), [data])
   127→   127→
   128→   128→  const rsiEnabled = indicators.some((i) => i.type === 'rsi' && i.enabled)
   129→   129→  const macdEnabled = indicators.some((i) => i.type === 'macd' && i.enabled)
   130→   130→  const bbEnabled = indicators.some((i) => i.type === 'bb' && i.enabled)
   131→   131→  const volEnabled = indicators.some((i) => i.type === 'volume' && i.enabled)
   132→   132→
   133→   133→  const layout = useMemo(
   134→   134→    () => computeLayout(width, height, rsiEnabled, macdEnabled, dpr),
   135→   135→    [width, height, rsiEnabled, macdEnabled]
   136→   136→  )
   137→   137→
   138→   138→  const viewport = useMemo(
   139→   139→    () => computeViewport(data, visibleStart, visibleEnd, indicatorData),
   140→   140→    [data, visibleStart, visibleEnd, indicatorData]
   141→   141→  )
   142→   142→
   143→   143→  // ==================== Master Render ====================
   144→   144→
   145→   145→  const render = useCallback(() => {
   146→   146→    const canvas = canvasRef.current
   147→   147→    if (!canvas) return
   148→   148→    const ctx = canvas.getContext('2d')
   149→   149→    if (!ctx) return
   150→   150→
   151→   151→    const w = width * dpr
   152→   152→    const h = height * dpr
   153→   153→    ctx.clearRect(0, 0, w, h)
   154→   154→    ctx.scale(dpr, dpr)
   155→   155→
   156→   156→    // Background
   157→   157→    ctx.fillStyle = themeRef.current.bg
   158→   158→    ctx.fillRect(0, 0, width, height)
   159→   159→
   160→   160→    // Grid
   161→   161→    drawGrid(ctx, viewport, layout)
   162→   162→
   163→   163→    // Volume
   164→   164→    if (volEnabled) drawVolume(ctx, data, viewport, layout)
   165→   165→
   166→   166→    // Candles
   167→   167→    drawCandles(ctx, data, viewport, layout)
   168→   168→
   169→   169→    // Overlay indicators
   170→   170→    drawOverlays(ctx, indicatorData, viewport, layout, bbEnabled)
   171→   171→
   172→   172→    // Sub-indicators
   173→   173→    if (rsiEnabled && layout.rsiTop != null && layout.rsiBottom != null)
   174→   174→      drawRSI(ctx, indicatorData, viewport, layout)
   175→   175→    if (macdEnabled && layout.macdTop != null && layout.macdBottom != null)
   176→   176→      drawMACD(ctx, indicatorData, viewport, layout)
   177→   177→
   178→   178→    // Separator lines for sub-charts
   179→   179→    drawSeparators(ctx, layout)
   180→   180→
   181→   181→    // Drawings
   182→   182→    renderDrawings(ctx, drawings, viewport, layout)
   183→   183→
   184→   184→    // Active preview
   185→   185→    if (dragRef.current) renderPreview(ctx, dragRef.current, viewport, layout)
   186→   186→
   187→   187→    // Crosshair + floating OHLC panel
   188→   188→    if (mouseRef.current && !dragRef.current && !panRef.current) {
   189→   189→      drawCrosshair(ctx, mouseRef.current, viewport, layout)
   190→   190→      drawInfoPanel(ctx, mouseRef.current, viewport, layout, data)
   191→   191→    }
   192→   192→
   193→   193→    // Price axis labels
   194→   194→    drawPriceAxis(ctx, viewport, layout)
   195→   195→
   196→   196→    // Time axis labels
   197→   197→    drawTimeAxis(ctx, data, viewport, layout)
   198→   198→
   199→   199→    // Last price marker
   200→   200→    drawLastPriceMarker(ctx, data, viewport, layout)
   201→   201→
   202→   202→    ctx.setTransform(1, 0, 0, 1, 0, 0)
   203→   203→  }, [data, width, height, dpr, viewport, layout, theme, indicatorData, drawings, rsiEnabled, macdEnabled, bbEnabled, volEnabled])
   204→   204→
   205→   205→  useEffect(() => {
   206→   206→    cancelAnimationFrame(animRef.current)
   207→   207→    animRef.current = requestAnimationFrame(render)
   208→   208→    return () => cancelAnimationFrame(animRef.current)
   209→   209→  }, [render])
   210→   210→
   211→   211→  // Resize
   212→   212→  useEffect(() => {
   213→   213→    let rafId = 0
   214→   214→    const onResize = () => { cancelAnimationFrame(rafId); rafId = requestAnimationFrame(render) }
   215→   215→    window.addEventListener('resize', onResize)
   216→   216→    return () => { window.removeEventListener('resize', onResize); cancelAnimationFrame(rafId) }
   217→   217→  }, [render])
   218→   218→
   219→   219→  // ==================== Input Handlers ====================
   220→   220→
   221→   221→  const getPos = useCallback((e: React.MouseEvent | React.TouchEvent) => {
   222→   222→    const canvas = canvasRef.current
   223→   223→    if (!canvas) return null
   224→   224→    const rect = canvas.getBoundingClientRect()
   225→   225→    if ('touches' in e) {
   226→   226→      const t = e.touches[0] || e.changedTouches[0]
   227→   227→      return t ? { x: t.clientX - rect.left, y: t.clientY - rect.top } : null
   228→   228→    }
   229→   229→    return { x: e.clientX - rect.left, y: e.clientY - rect.top }
   230→   230→  }, [])
   231→   231→
   232→   232→  const onMouseDown = useCallback((e: React.MouseEvent) => {
   233→   233→    const pos = getPos(e)
   234→   234→    if (!pos) return
   235→   235→
   236→   236→    // Middle-click or alt+click = pan
   237→   237→    if (e.button === 1 || (e.button === 0 && e.altKey)) {
   238→   238→      panRef.current = { startX: pos.x, startOffset: visibleEnd - visibleStart }
   239→   239→      return
   240→   240→    }
   241→   241→
   242→   242→    if (activeTool !== 'crosshair' && pos.x < layout.chartRight) {
   243→   243→      const idx = xToIndex(pos.x, viewport, layout)
   244→   244→      const price = yToPrice(pos.y, viewport, layout)
   245→   245→      dragRef.current = {
   246→   246→        type: activeTool,
   247→   247→        startScreen: pos,
   248→   248→        currentScreen: pos,
   249→   249→        startData: { index: idx, price },
   250→   250→        currentData: { index: idx, price },
   251→   251→      }
   252→   252→    }
   253→   253→  }, [activeTool, viewport, layout, getPos, visibleStart, visibleEnd])
   254→   254→
   255→   255→  const onMouseMove = useCallback((e: React.MouseEvent) => {
   256→   256→    const pos = getPos(e)
   257→   257→    if (!pos) return
   258→   258→    mouseRef.current = pos
   259→   259→
   260→   260→    if (panRef.current) {
   261→   261→      const dx = pos.x - panRef.current.startX
   262→   262→      const cpi = (visibleEnd - visibleStart) / (layout.chartRight - layout.chartLeft)
   263→   263→      const shift = Math.round(-dx * cpi)
   264→   264→      let ns = Math.max(0, visibleStart + shift)
   265→   265→      let ne = ns + panRef.current.startOffset
   266→   266→      if (ne >= data.length) { ne = data.length - 1; ns = Math.max(0, ne - panRef.current.startOffset) }
   267→   267→      setVisibleStart(ns); setVisibleEnd(ne)
   268→   268→      return
   269→   269→    }
   270→   270→
   271→   271→    if (dragRef.current) {
   272→   272→      dragRef.current = {
   273→   273→        ...dragRef.current,
   274→   274→        currentScreen: pos,
   275→   275→        currentData: { index: xToIndex(pos.x, viewport, layout), price: yToPrice(pos.y, viewport, layout) },
   276→   276→      }
   277→   277→      render(); return
   278→   278→    }
   279→   279→
   280→   280→    if (onCrosshairMove && pos.x < layout.chartRight && pos.y < layout.chartBottom) {
   281→   281→      const idx = xToIndex(pos.x, viewport, layout)
   282→   282→      const price = yToPrice(pos.y, viewport, layout)
   283→   283→      const candle = data[idx] ?? null
   284→   284→      onCrosshairMove(price, idx, candle)
   285→   285→    }
   286→   286→    render()
   287→   287→  }, [viewport, layout, getPos, render, onCrosshairMove, data, visibleStart, visibleEnd])
   288→   288→
   289→   289→  const onMouseUp = useCallback(() => {
   290→   290→    if (panRef.current) { panRef.current = null; return }
   291→   291→    if (dragRef.current) { finalizeDrawing(); dragRef.current = null; render() }
   292→   292→  }, [render])
   293→   293→
   294→   294→  const onMouseLeave = useCallback(() => {
   295→   295→    mouseRef.current = null
   296→   296→    if (panRef.current) panRef.current = null
   297→   297→    if (dragRef.current) { finalizeDrawing(); dragRef.current = null }
   298→   298→    render()
   299→   299→    onCrosshairMove?.(null, null, null)
   300→   300→  }, [render, onCrosshairMove])
   301→   301→
   302→   302→  // Touch
   303→   303→  const onTouchStart = useCallback((e: React.TouchEvent) => {
   304→   304→    if (e.touches.length !== 1) return
   305→   305→    const pos = getPos(e)
   306→   306→    if (!pos) return
   307→   307→    mouseRef.current = pos
   308→   308→    if (activeTool !== 'crosshair' && pos.x < layout.chartRight) {
   309→   309→      const idx = xToIndex(pos.x, viewport, layout)
   310→   310→      const price = yToPrice(pos.y, viewport, layout)
   311→   311→      dragRef.current = { type: activeTool, startScreen: pos, currentScreen: pos, startData: { index: idx, price }, currentData: { index: idx, price } }
   312→   312→    } else {
   313→   313→      panRef.current = { startX: pos.x, startOffset: visibleEnd - visibleStart }
   314→   314→    }
   315→   315→  }, [activeTool, viewport, layout, getPos, visibleStart, visibleEnd])
   316→   316→
   317→   317→  const onTouchMove = useCallback((e: React.TouchEvent) => {
   318→   318→    e.preventDefault()
   319→   319→    const pos = getPos(e)
   320→   320→    if (!pos) return
   321→   321→    mouseRef.current = pos
   322→   322→    if (panRef.current && !dragRef.current) {
   323→   323→      const dx = pos.x - panRef.current.startX
   324→   324→      const cpi = (visibleEnd - visibleStart) / (layout.chartRight - layout.chartLeft)
   325→   325→      const shift = Math.round(-dx * cpi)
   326→   326→      let ns = Math.max(0, visibleStart + shift)
   327→   327→      let ne = ns + panRef.current.startOffset
   328→   328→      if (ne >= data.length) { ne = data.length - 1; ns = Math.max(0, ne - panRef.current.startOffset) }
   329→   329→      setVisibleStart(ns); setVisibleEnd(ne)
   330→   330→    } else if (dragRef.current) {
   331→   331→      dragRef.current = {
   332→   332→        ...dragRef.current,
   333→   333→        currentScreen: pos,
   334→   334→        currentData: { index: xToIndex(pos.x, viewport, layout), price: yToPrice(pos.y, viewport, layout) },
   335→   335→      }
   336→   336→    }
   337→   337→    render()
   338→   338→  }, [viewport, layout, getPos, render, visibleStart, visibleEnd])
   339→   339→
   340→   340→  const onTouchEnd = useCallback(() => {
   341→   341→    mouseRef.current = null
   342→   342→    if (panRef.current) panRef.current = null
   343→   343→    if (dragRef.current) { finalizeDrawing(); dragRef.current = null }
   344→   344→    render()
   345→   345→  }, [render])
   346→   346→
   347→   347→  // Wheel zoom
   348→   348→  const onWheel = useCallback((e: React.WheelEvent) => {
   349→   349→    e.preventDefault()
   350→   350→    const factor = e.deltaY > 0 ? 1.12 : 0.89
   351→   351→    const range = visibleEnd - visibleStart
   352→   352→    const newRange = Math.round(Math.max(15, Math.min(data.length - 1, range * factor)))
   353→   353→    const center = Math.round((visibleStart + visibleEnd) / 2)
   354→   354→    let ns = Math.max(0, center - Math.round(newRange / 2))
   355→   355→    let ne = ns + newRange - 1
   356→   356→    if (ne >= data.length) { ne = data.length - 1; ns = Math.max(0, ne - newRange + 1) }
   357→   357→    setVisibleStart(ns); setVisibleEnd(ne)
   358→   358→  }, [visibleStart, visibleEnd, data.length])
   359→   359→
   360→   360→  // Context menu → alert
   361→   361→  const onContextMenu = useCallback((e: React.MouseEvent) => {
   362→   362→    const pos = getPos(e)
   363→   363→    if (!pos) return
   364→   364→    for (const d of drawings) {
   365→   365→      if (d.type === 'hline' && d.points.length > 0) {
   366→   366→        const ly = priceToY(d.points[0].price, viewport, layout)
   367→   367→        if (Math.abs(pos.y - ly) < 10 && pos.x < layout.chartRight) { e.preventDefault(); onDrawingClick?.(d); return }
   368→   368→      }
   369→   369→      if (d.type === 'trendline' && d.points.length >= 2) {
   370→   370→        const x1 = indexToX(d.points[0].index, viewport, layout), y1 = priceToY(d.points[0].price, viewport, layout)
   371→   371→        const x2 = indexToX(d.points[1].index, viewport, layout), y2 = priceToY(d.points[1].price, viewport, layout)
   372→   372→        if (ptLineDist(pos.x, pos.y, x1, y1, x2, y2) < 12) { e.preventDefault(); onDrawingClick?.(d); return }
   373→   373→      }
   374→   374→    }
   375→   375→  }, [drawings, viewport, layout, getPos, onDrawingClick])
   376→   376→
   377→   377→  // Double-click delete
   378→   378→  const onDoubleClick = useCallback((e: React.MouseEvent) => {
   379→   379→    const pos = getPos(e)
   380→   380→    if (!pos) return
   381→   381→    for (let i = drawings.length - 1; i >= 0; i--) {
   382→   382→      const d = drawings[i]
   383→   383→      if (d.type === 'hline' && d.points.length > 0) {
   384→   384→        if (Math.abs(priceToY(d.points[0].price, viewport, layout) - pos.y) < 10 && pos.x < layout.chartRight)
   385→   385→          { onDrawingsChange(drawings.filter((_, j) => j !== i)); return }
   386→   386→      }
   387→   387→      if (d.type === 'trendline' && d.points.length >= 2) {
   388→   388→        const x1 = indexToX(d.points[0].index, viewport, layout), y1 = priceToY(d.points[0].price, viewport, layout)
   389→   389→        const x2 = indexToX(d.points[1].index, viewport, layout), y2 = priceToY(d.points[1].price, viewport, layout)
   390→   390→        if (ptLineDist(pos.x, pos.y, x1, y1, x2, y2) < 12)
   391→   391→          { onDrawingsChange(drawings.filter((_, j) => j !== i)); return }
   392→   392→      }
   393→   393→      if (d.type === 'fibonacci' && d.points.length >= 2) {
   394→   394→        const lx1 = Math.min(indexToX(d.points[0].index, viewport, layout), indexToX(d.points[1].index, viewport, layout))
   395→   395→        const lx2 = Math.max(indexToX(d.points[0].index, viewport, layout), indexToX(d.points[1].index, viewport, layout))
   396→   396→        if (pos.x >= lx1 && pos.x <= lx2) { onDrawingsChange(drawings.filter((_, j) => j !== i)); return }
   397→   397→      }
   398→   398→      if (d.type === 'rectangle' && d.points.length >= 2) {
   399→   399→        const rx1 = Math.min(indexToX(d.points[0].index, viewport, layout), indexToX(d.points[1].index, viewport, layout))
   400→   400→        const rx2 = Math.max(indexToX(d.points[0].index, viewport, layout), indexToX(d.points[1].index, viewport, layout))
   401→   401→        const ry1 = Math.min(priceToY(d.points[0].price, viewport, layout), priceToY(d.points[1].price, viewport, layout))
   402→   402→        const ry2 = Math.max(priceToY(d.points[0].price, viewport, layout), priceToY(d.points[1].price, viewport, layout))
   403→   403→        if (pos.x >= rx1 && pos.x <= rx2 && pos.y >= ry1 && pos.y <= ry2)
   404→   404→          { onDrawingsChange(drawings.filter((_, j) => j !== i)); return }
   405→   405→      }
   406→   406→    }
   407→   407→  }, [drawings, viewport, layout, getPos, onDrawingsChange])
   408→   408→
   409→   409→  const finalizeDrawing = useCallback(() => {
   410→   410→    if (!dragRef.current?.startData || !dragRef.current?.currentData) return
   411→   411→    const { type, startData, currentData } = dragRef.current
   412→   412→    const color = DRAW_COLORS[colorIdx % DRAW_COLORS.length]
   413→   413→    const nd: ChartDrawing = { id: `d-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`, type, color, width: 1.5, points: [startData, currentData] }
   414→   414→    if (type === 'hline') nd.points = [startData]

   450→   450→function drawGrid(ctx: CanvasRenderingContext2D, vp: ChartViewport, layout: ChartLayout) {
   451→   451→  const { chartTop, chartBottom, chartRight, volumeHeight } = layout
   452→   452→  const priceBottom = chartBottom - volumeHeight
   453→   453→  const priceRange = vp.maxPrice - vp.minPrice
   454→   454→  const step = niceStep(priceRange, 6)
   455→   455→
   456→   456→  // Horizontal price grid
   457→   457→  let price = Math.ceil(vp.minPrice / step) * step
   458→   458→  while (price <= vp.maxPrice) {
   459→   459→    const y = Math.round(priceToY(price, vp, layout)) + 0.5
   460→   460→    if (y >= chartTop && y <= priceBottom) {
   461→   461→      ctx.strokeStyle = gridColor(y, chartTop, priceBottom)
   462→   462→      ctx.lineWidth = 1
   463→   463→      ctx.beginPath(); ctx.moveTo(layout.chartLeft, y); ctx.lineTo(chartRight, y); ctx.stroke()
   464→   464→    }
   465→   465→    price += step
   466→   466→  }
   467→   467→
   468→   468→  // Vertical time grid
   469→   469→  const total = vp.endIndex - vp.startIndex + 1
   470→   470→  const stepC = Math.max(1, Math.round(total / 8))
   471→   471→  for (let i = vp.startIndex; i <= vp.endIndex; i += stepC) {
   472→   472→    const x = Math.round(indexToX(i, vp, layout)) + 0.5
   473→   473→    ctx.strokeStyle = 'rgba(128,128,128,0.08)'
   474→   474→    ctx.lineWidth = 1
   475→   475→    ctx.beginPath(); ctx.moveTo(x, chartTop); ctx.lineTo(x, chartBottom); ctx.stroke()
   476→   476→  }
   477→   477→
   478→   478→  // Chart border
   479→   479→  ctx.strokeStyle = 'rgba(128,128,128,0.12)'
   480→   480→  ctx.lineWidth = 1
   481→   481→  ctx.strokeRect(layout.chartLeft + 0.5, chartTop + 0.5, chartRight - layout.chartLeft - 1, chartBottom - chartTop - 1)
   482→   482→}

   484→   484→function gridColor(y: number, top: number, bottom: number) {
   485→   485→  const mid = (top + bottom) / 2
   486→   486→  const dist = Math.abs(y - mid) / (bottom - top)
   487→   487→  return dist < 0.25 ? 'rgba(128,128,128,0.06)' : 'rgba(128,128,128,0.12)'
   488→   488→}

   490→   490→function drawCandles(ctx: CanvasRenderingContext2D, data: OHLCData[], vp: ChartViewport, layout: ChartLayout) {
   491→   491→  const { chartRight, chartBottom, volumeHeight } = layout
   492→   492→  const priceBottom = chartBottom - volumeHeight
   493→   493→  const total = vp.endIndex - vp.startIndex + 1
   494→   494→  const chartW = chartRight - layout.chartLeft
   495→   495→  const slotW = chartW / total
   496→   496→  const bodyW = Math.max(2, slotW * 0.75)
   497→   497→  const r = Math.min(2, bodyW / 3) // corner radius
   498→   498→
   499→   499→  for (let i = vp.startIndex; i <= vp.endIndex; i++) {
   500→   500→    const c = data[i]
   501→   501→    if (!c) continue
   502→   502→    const x = indexToX(i, vp, layout)
   503→   503→    const green = c.close >= c.open
   504→   504→
   505→   505→    // Wick - rounded cap via lineCap
   506→   506→    ctx.lineCap = 'round'
   507→   507→    ctx.strokeStyle = green ? themeRef.current.gain : themeRef.current.loss
   508→   508→    ctx.lineWidth = Math.max(1.5, slotW * 0.15)
   509→   509→    ctx.beginPath()
   510→   510→    ctx.moveTo(x, Math.round(priceToY(c.high, vp, layout)))
   511→   511→    ctx.lineTo(x, Math.round(priceToY(c.low, vp, layout)))
   512→   512→    ctx.stroke()
   513→   513→    ctx.lineCap = 'butt'
   514→   514→
   515→   515→    // Body with rounded corners
   516→   516→    let top = priceToY(Math.max(c.open, c.close), vp, layout)
   517→   517→    let bot = priceToY(Math.min(c.open, c.close), vp, layout)
   518→   518→    let bodyH = Math.max(2, bot - top)
   519→   519→
   520→   520→    ctx.fillStyle = green ? themeRef.current.gain : themeRef.current.loss
   521→   521→    if (r >= 1 && bodyH > r * 2) {
   522→   522→      roundedRect(ctx, x - bodyW / 2, top, bodyW, bodyH, r)
   523→   523→      ctx.fill()
   524→   524→    } else {
   525→   525→      ctx.fillRect(x - bodyW / 2, Math.round(top), bodyW, Math.round(bodyH))
   526→   526→    }
   527→   527→
   528→   528→    // Subtle glow for last candle
   529→   529→    if (i === vp.endIndex) {
   530→   530→      ctx.shadowColor = green ? themeRef.current.gain : themeRef.current.loss
   531→   531→      ctx.shadowBlur = 8
   532→   532→      ctx.fillRect(x - bodyW / 2, Math.round(top), bodyW, Math.round(bodyH))
   533→   533→      ctx.shadowBlur = 0
   534→   534→      ctx.shadowColor = 'transparent'
   535→   535→    }
   536→   536→  }
   537→   537→}

   539→   539→function drawVolume(ctx: CanvasRenderingContext2D, data: OHLCData[], vp: ChartViewport, layout: ChartLayout) {
   540→   540→  const { chartBottom, volumeHeight, chartRight } = layout
   541→   541→  const total = vp.endIndex - vp.startIndex + 1
   542→   542→  const chartW = chartRight - layout.chartLeft
   543→   543→  const barW = chartW / total
   544→   544→  let maxVol = 0
   545→   545→  for (let i = vp.startIndex; i <= vp.endIndex; i++) maxVol = Math.max(maxVol, data[i].volume)
   546→   546→  if (maxVol === 0) return
   547→   547→
   548→   548→  // Volume separator
   549→   549→  const volTop = chartBottom - volumeHeight
   550→   550→  ctx.strokeStyle = 'rgba(128,128,128,0.1)'
   551→   551→  ctx.lineWidth = 0.5
   552→   552→  ctx.beginPath(); ctx.moveTo(layout.chartLeft, Math.round(volTop) + 0.5); ctx.lineTo(chartRight, Math.round(volTop) + 0.5); ctx.stroke()
   553→   553→
   554→   554→  for (let i = vp.startIndex; i <= vp.endIndex; i++) {
   555→   555→    const c = data[i]
   556→   556→    const x = indexToX(i, vp, layout)
   557→   557→    const green = c.close >= c.open
   558→   558→    const barH = (c.volume / maxVol) * volumeHeight * 0.85
   559→   559→    const bw = Math.max(1, barW * 0.55)
   560→   560→
   561→   561→    // Gradient fill
   562→   562→    const grad = ctx.createLinearGradient(0, chartBottom - barH, 0, chartBottom)
   563→   563→    grad.addColorStop(0, green ? IND_COLORS.volGain : IND_COLORS.volLoss)
   564→   564→    grad.addColorStop(1, 'transparent')
   565→   565→    ctx.fillStyle = grad
   566→   566→    ctx.fillRect(x - bw / 2, chartBottom - barH, bw, barH)
   567→   567→  }

   570→   570→function drawOverlays(ctx: CanvasRenderingContext2D, ind: IndicatorData, vp: ChartViewport, layout: ChartLayout, bb: boolean) {
   571→   571→  // MA lines
   572→   572→  const maList = [
   573→   573→    { d: ind.ma20, color: IND_COLORS.ma20 },
   574→   574→    { d: ind.ma50, color: IND_COLORS.ma50 },
   575→   575→    { d: ind.ma100, color: IND_COLORS.ma100 },
   576→   576→  ]
   577→   577→  for (const ma of maList) {
   578→   578→    if (!ma.d) continue

   581→   581→    ctx.globalAlpha = 1

   583→   583→    let started = false
   584→   584→    for (let i = vp.startIndex; i <= vp.endIndex; i++) {
   585→   585→      const v = ma.d[i]
   586→   586→      if (isNaN(v)) continue
   587→   587→      const x = indexToX(i, vp, layout)
   588→   588→      const y = priceToY(v, vp, layout)
   589→   589→      if (!started) { ctx.moveTo(x, y); started = true } else ctx.lineTo(x, y)

   592→   592→    ctx.globalAlpha = 1
   593→   593→  }

   595→   595→  // Bollinger Bands

   597→   597→    // Fill
   598→   598→    ctx.fillStyle = IND_COLORS.bbFill
   599→   599→    ctx.beginPath()
   600→   600→    let s = false
   601→   601→    for (let i = vp.startIndex; i <= vp.endIndex; i++) {
   602→   602→      const v = ind.bbUpper[i]; if (isNaN(v)) continue
   603→   603→      if (!s) { ctx.moveTo(indexToX(i, vp, layout), priceToY(v, vp, layout)); s = true }
   604→   604→      else ctx.lineTo(indexToX(i, vp, layout), priceToY(v, vp, layout))
   605→   605→    }
   606→   606→    for (let i = vp.endIndex; i >= vp.startIndex; i--) {
   607→   607→      const v = ind.bbLower[i]; if (isNaN(v)) continue
   608→   608→      ctx.lineTo(indexToX(i, vp, layout), priceToY(v, vp, layout))
   609→   609→    }
   610→   610→    ctx.closePath(); ctx.fill()
   611→   611→
   612→   612→    // Lines
   613→   613→    const bbLines = [ind.bbUpper, ind.bbMiddle, ind.bbLower]
   614→   614→    for (const line of bbLines) {
   615→   615→      if (!line) continue
   616→   616→      ctx.strokeStyle = IND_COLORS.bbLine; ctx.lineWidth = 0.8; ctx.setLineDash([4, 4])
   617→   617→      ctx.beginPath(); s = false
   618→   618→      for (let i = vp.startIndex; i <= vp.endIndex; i++) {
   619→   619→        const v = line[i]; if (isNaN(v)) continue
   620→   620→        const x = indexToX(i, vp, layout), y = priceToY(v, vp, layout)
   621→   621→        if (!s) { ctx.moveTo(x, y); s = true } else ctx.lineTo(x, y)
   622→   622→      }
   623→   623→      ctx.stroke(); ctx.setLineDash([])
   624→   624→    }
   625→   625→  }

   670→   670→function drawMACD(ctx: CanvasRenderingContext2D, ind: IndicatorData, vp: ChartViewport, layout: ChartLayout) {

   676→   676→
   677→   677→  let mn = Infinity, mx = -Infinity

   700→   700→    ctx.fillRect(Math.round(x - bw * 0.25), Math.round(Math.min(y, zy)), Math.round(bw * 0.5), Math.round(bh))
   701→   701→  }
   702→   702→
   703→   703→  // Lines

   705→   705→    if (!arr) return
   706→   706→    ctx.strokeStyle = color; ctx.lineWidth = 1.2; ctx.beginPath()
   707→   707→    let s = false
   708→   708→    for (let i = vp.startIndex; i <= vp.endIndex; i++) {
   709→   709→      const v = arr[i]; if (isNaN(v)) continue
   710→   710→      const x = indexToX(i, vp, layout), y = toY(v)
   711→   711→      if (!s) { ctx.moveTo(x, y); s = true } else ctx.lineTo(x, y)
   712→   712→    }
   713→   713→    ctx.stroke()
   714→   714→  }
   715→   715→  drawLine(ind.macdLine, IND_COLORS.macdLine)
   716→   716→  drawLine(ind.macdSignal, IND_COLORS.macdSignal)
   717→   717→
   718→   718→  ctx.fillStyle = 'rgba(128,128,128,0.4)'; ctx.font = '10px -apple-system,system-ui,sans-serif'

   864→   864→function renderDrawings(ctx: CanvasRenderingContext2D, drawings: ChartDrawing[], vp: ChartViewport, layout: ChartLayout) {
   865→   865→  for (const d of drawings) {
   866→   866→    ctx.strokeStyle = d.color; ctx.fillStyle = d.color; ctx.lineWidth = d.width
   867→   867→
   868→   868→    if (d.type === 'hline' && d.points.length > 0) {
   869→   869→      const y = priceToY(d.points[0].price, vp, layout)
   870→   870→      ctx.setLineDash([8, 4]); ctx.globalAlpha = 0.7
   871→   871→      ctx.beginPath(); ctx.moveTo(layout.chartLeft, Math.round(y) + 0.5); ctx.lineTo(layout.chartRight, Math.round(y) + 0.5); ctx.stroke()
   872→   872→      ctx.setLineDash([]); ctx.globalAlpha = 1
   873→   873→
   874→   874→      // Price tag on right
   875→   875→      ctx.fillStyle = d.color; roundedRect(ctx, layout.chartRight, y - 8, 52, 16, 3); ctx.fill()
   876→   876→      ctx.fillStyle = '#fff'; ctx.font = 'bold 9px -apple-system,system-ui,sans-serif'
   877→   877→      ctx.fillText(d.points[0].price.toFixed(2), layout.chartRight + 4, y + 3)
   878→   878→    }
   879→   879→
   880→   880→    if (d.type === 'trendline' && d.points.length >= 2) {
   881→   881→      const x1 = indexToX(d.points[0].index, vp, layout), y1 = priceToY(d.points[0].price, vp, layout)
   882→   882→      const x2 = indexToX(d.points[1].index, vp, layout), y2 = priceToY(d.points[1].price, vp, layout)
   883→   883→
   884→   884→      // Glow
   885→   885→      ctx.shadowColor = d.color; ctx.shadowBlur = 4
   886→   886→      ctx.beginPath(); ctx.moveTo(x1, y1); ctx.lineTo(x2, y2); ctx.stroke()
   887→   887→      ctx.shadowBlur = 0; ctx.shadowColor = 'transparent'
   888→   888→
   889→   889→      // Endpoint dots
   890→   890→      for (const [px, py] of [[x1, y1], [x2, y2]]) {
   891→   891→        ctx.fillStyle = d.color; ctx.beginPath(); ctx.arc(px, py, 3, 0, Math.PI * 2); ctx.fill()
   892→   892→        ctx.fillStyle = '#fff'; ctx.beginPath(); ctx.arc(px, py, 1.5, 0, Math.PI * 2); ctx.fill()
   893→   893→      }
   894→   894→    }
   895→   895→
   896→   896→    if (d.type === 'fibonacci' && d.points.length >= 2 && d.fibLevels) {
   897→   897→      const x1 = Math.min(indexToX(d.points[0].index, vp, layout), indexToX(d.points[1].index, vp, layout))
   898→   898→      const x2 = Math.max(indexToX(d.points[0].index, vp, layout), indexToX(d.points[1].index, vp, layout))
   899→   899→      const fibColors = ['rgba(239,83,80,0.06)', 'rgba(247,162,27,0.08)', 'rgba(126,87,194,0.08)', 'rgba(128,128,128,0.04)', 'rgba(126,87,194,0.08)', 'rgba(38,166,154,0.08)', 'rgba(38,166,154,0.06)']
   900→   900→      const fibLabels = ['0.0%', '23.6%', '38.2%', '50.0%', '61.8%', '78.6%', '100%']
   901→   901→
   902→   902→      for (let fi = 0; fi < d.fibLevels.length; fi++) {
   903→   903→        const price = d.fibLevels[fi]
   904→   904→        const y = priceToY(price, vp, layout)
   905→   905→
   906→   906→        // Fill
   907→   907→        if (fi < d.fibLevels.length - 1) {
   908→   908→          const ny = priceToY(d.fibLevels[fi + 1], vp, layout)
   909→   909→          ctx.fillStyle = fibColors[fi] ?? 'rgba(128,128,128,0.04)'
   910→   910→          ctx.fillRect(x1, y, x2 - x1, ny - y)
   911→   911→        }
   912→   912→
   913→   913→        // Line
   914→   914→        ctx.strokeStyle = 'rgba(128,128,128,0.2)'; ctx.lineWidth = 0.5; ctx.setLineDash([2, 2])
   915→   915→        ctx.beginPath(); ctx.moveTo(x1, Math.round(y) + 0.5); ctx.lineTo(x2, Math.round(y) + 0.5); ctx.stroke()
   916→   916→        ctx.setLineDash([])
   917→   917→
   918→   918→        // Label
   919→   919→        ctx.fillStyle = 'rgba(200,200,200,0.7)'; ctx.font = '9px -apple-system,system-ui,sans-serif'; ctx.textAlign = 'left'
   920→   920→        ctx.fillText(`${fibLabels[fi]}  ${price.toFixed(2)}`, x1 + 4, y - 3)
   921→   921→      }
   922→   922→    }
   923→   923→
   924→   924→    if (d.type === 'rectangle' && d.points.length >= 2) {
   925→   925→      const x1 = indexToX(d.points[0].index, vp, layout), y1 = priceToY(d.points[0].price, vp, layout)
   926→   926→      const x2 = indexToX(d.points[1].index, vp, layout), y2 = priceToY(d.points[1].price, vp, layout)
   927→   927→      const rx = Math.min(x1, x2), ry = Math.min(y1, y2), rw = Math.abs(x2 - x1), rh = Math.abs(y2 - y1)
   928→   928→
   929→   929→      ctx.fillStyle = d.color.replace(')', ',0.06)').replace('rgb', 'rgba').replace('oklch', 'rgba')
   930→   930→      ctx.fillRect(rx, ry, rw, rh)
   931→   931→      ctx.strokeStyle = d.color; ctx.lineWidth = 1.5
   932→   932→      ctx.strokeRect(rx, ry, rw, rh)
   933→   933→    }
   934→   934→  }
   935→   935→}

   939→   939→function renderPreview(ctx: CanvasRenderingContext2D, drawing: ActiveDrawing, vp: ChartViewport, layout: ChartLayout) {
   940→   940→  if (!drawing.startData || !drawing.currentData) return
   941→   941→  ctx.strokeStyle = DRAW_COLORS[0]; ctx.lineWidth = 1.5; ctx.setLineDash([5, 4]); ctx.globalAlpha = 0.6
   942→   942→
   943→   943→  if (drawing.type === 'hline') {
   944→   944→    const y = priceToY(drawing.startData.price, vp, layout)
   945→   945→    ctx.beginPath(); ctx.moveTo(layout.chartLeft, Math.round(y) + 0.5); ctx.lineTo(layout.chartRight, Math.round(y) + 0.5); ctx.stroke()
   946→   946→  }
   947→   947→
   948→   948→  if (drawing.type === 'trendline') {
   949→   949→    ctx.beginPath()
   950→   950→    ctx.moveTo(indexToX(drawing.startData.index, vp, layout), priceToY(drawing.startData.price, vp, layout))
   951→   951→    ctx.lineTo(indexToX(drawing.currentData.index, vp, layout), priceToY(drawing.currentData.price, vp, layout))
   952→   952→    ctx.stroke()
   953→   953→  }
   954→   954→
   955→   955→  if (drawing.type === 'fibonacci') {
   956→   956→    const lx1 = Math.min(indexToX(drawing.startData.index, vp, layout), indexToX(drawing.currentData.index, vp, layout))
   957→   957→    const lx2 = Math.max(indexToX(drawing.startData.index, vp, layout), indexToX(drawing.currentData.index, vp, layout))
   958→   958→    const hi = Math.max(drawing.startData.price, drawing.currentData.price)
   959→   959→    const lo = Math.min(drawing.startData.price, drawing.currentData.price)
   960→   960→    const levels = fibonacciLevels(hi, lo)
   961→   961→    for (const p of levels) {
   962→   962→      const y = priceToY(p, vp, layout)
   963→   963→      ctx.strokeStyle = 'rgba(200,200,200,0.3)'; ctx.lineWidth = 0.5
   964→   964→      ctx.beginPath(); ctx.moveTo(lx1, Math.round(y) + 0.5); ctx.lineTo(lx2, Math.round(y) + 0.5); ctx.stroke()
   965→   965→    }
   966→   966→  }
   967→   967→
   968→   968→  if (drawing.type === 'rectangle') {
   969→   969→    const x1 = indexToX(drawing.startData.index, vp, layout), y1 = priceToY(drawing.startData.price, vp, layout)
   970→   970→    const x2 = indexToX(drawing.currentData.index, vp, layout), y2 = priceToY(drawing.currentData.price, vp, layout)
   971→   971→    ctx.strokeRect(Math.min(x1, x2), Math.min(y1, y2), Math.abs(x2 - x1), Math.abs(y2 - y1))
   972→   972→  }
   973→   973→
   974→   974→  ctx.setLineDash([]); ctx.globalAlpha = 1
   975→   975→}

   979→   979→function roundedRect(ctx: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number) {
   980→   980→  r = Math.min(r, w / 2, h / 2)
   981→   981→  ctx.beginPath()
   982→   982→  ctx.moveTo(x + r, y)
   983→   983→  ctx.lineTo(x + w - r, y); ctx.arcTo(x + w, y, x + w, y + r, r)
   984→   984→  ctx.lineTo(x + w, y + h - r); ctx.arcTo(x + w, y + h, x + w - r, y + h, r)
   985→   985→  ctx.lineTo(x + r, y + h); ctx.arcTo(x, y + h, x, y + h - r, r)
   986→   986→  ctx.lineTo(x, y + r); ctx.arcTo(x, y, x + r, y, r)
   987→   987→  ctx.closePath()
   988→   988→}

   990→   990→function ptLineDist(px: number, py: number, x1: number, y1: number, x2: number, y2: number): number {
   991→   991→  const dx = x2 - x1, dy = y2 - y1
   992→   992→  const lenSq = dx * dx + dy * dy
   993→   993→  if (lenSq === 0) return Math.hypot(px - x1, py - y1)
   994→   994→  let t = ((px - x1) * dx + (py - y1) * dy) / lenSq
   995→   995→  t = Math.max(0, Math.min(1, t))
   996→   996→  return Math.hypot(px - (x1 + t * dx), py - (y1 + t * dy))
   997→   997→}
