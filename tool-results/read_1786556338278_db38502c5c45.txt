   108→function formatPrice(price: number, symbol: string): string {
   109→  if (['BTC', 'ETH', 'BNB', 'SOL', 'XRP', 'ADA'].includes(symbol)) {
   110→    return `$${price.toLocaleString('en-US', { maximumFractionDigits: 2 })}`
   111→  }
   112→  if (['SJC', 'XAU'].includes(symbol)) {
   113→    return `${(price / 1_000_000).toFixed(1)} tr VNĐ`
   114→  }
   115→  return `${price.toLocaleString('vi-VN')} VNĐ`
   116→}

   118→// ---------- Step 1: Asset Selection ----------
   119→function StepAsset({
   120→  selected,
   121→  onSelect,
   122→}: {
   123→  selected: Asset | null
   124→  onSelect: (asset: Asset) => void
   125→}) {
   126→  const [search, setSearch] = useState('')
   127→  const [assetType, setAssetType] = useState<string>('all')
   128→  const [assets, setAssets] = useState<Asset[]>([])
   129→  const [loading, setLoading] = useState(false)
   130→
   131→  const fetchAssets = useCallback(async (query?: string, type?: string) => {
   132→    setLoading(true)
   133→    try {
   134→      const params = new URLSearchParams()
   135→      if (query) params.set('search', query)
   136→      if (type && type !== 'all') params.set('type', type)
   137→      const res = await fetch(`/api/assets?${params.toString()}`)
   138→      const json = await res.json()
   139→      setAssets(json.data ?? [])
   140→    } catch {
   141→      setAssets([])
   142→    } finally {
   143→      setLoading(false)
   144→    }
   145→  }, [])
   146→
   147→  useEffect(() => {
   148→    fetchAssets(search, assetType)
   149→  }, [search, assetType, fetchAssets])
   150→
   151→  const handleRecentClick = (symbol: string) => {
   152→    const asset = assets.find((a) => a.symbol === symbol)
   153→    if (asset) onSelect(asset)
   154→    else {
   155→      fetch(`/api/assets?search=${symbol}`)
   156→        .then((r) => r.json())
   157→        .then((json) => {
   158→          const found = (json.data ?? []).find((a: Asset) => a.symbol === symbol)
   159→          if (found) onSelect(found)
   160→        })
   161→        .catch(() => {})
   162→    }
   163→  }
   164→
   165→  return (
   166→    <motion.div
   167→      initial={{ opacity: 0, x: 20 }}
   168→      animate={{ opacity: 1, x: 0 }}
   169→      exit={{ opacity: 0, x: -20 }}
   170→      className="space-y-4"
   171→    >
   172→      {/* Search - full width, proper height */}
   173→      <div className="relative">
   174→        <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   175→        <Input
   176→          placeholder="Tìm mã tài sản..."
   177→          value={search}
   178→          onChange={(e) => setSearch(e.target.value)}
   179→          className="h-12 pl-9 text-sm sm:h-11"
   180→        />
   181→      </div>
   182→
   183→      {/* Asset type filter - horizontal scrollable chips */}
   184→      <div className="flex flex-nowrap gap-1.5 overflow-x-auto scrollbar-none">
   185→        {ASSET_TYPES.map((t) => (
   186→          <button
   187→            key={t.key}
   188→            onClick={() => setAssetType(t.key)}
   189→            className={cn(
   190→              'flex min-h-[36px] shrink-0 items-center rounded-full border px-3 text-xs font-medium transition-colors',
   191→              assetType === t.key
   192→                ? 'border-primary bg-primary/10 text-primary'
   193→                : 'border-border text-muted-foreground hover:bg-accent'
   194→            )}
   195→          >
   196→            {t.label}
   197→          </button>
   198→        ))}
   199→      </div>
   200→
   201→      {/* Selected asset */}
   202→      {selected && (
   203→        <div className="rounded-lg border border-primary/30 bg-primary/5 p-3">
   204→          <div className="flex items-center justify-between">
   205→            <div>
   206→              <span className="font-semibold">{selected.symbol}</span>
   207→              <span className="ml-2 text-sm text-muted-foreground">{selected.name}</span>
   208→            </div>
   209→            <button onClick={() => onSelect(null as unknown as Asset)} className="flex min-h-[44px] min-w-[44px] items-center justify-center text-muted-foreground hover:text-foreground">
   210→              <X className="h-4 w-4" />
   211→            </button>
   212→          </div>
   213→          <div className="mt-1 flex items-center gap-2">
   214→            <span className="text-lg font-bold">{formatPrice(selected.price, selected.symbol)}</span>
   215→            <span className={cn('text-sm font-medium', selected.change24h >= 0 ? 'text-gain' : 'text-loss')}>
   216→              {selected.change24h >= 0 ? '+' : ''}{selected.changePercent.toFixed(2)}%
   217→            </span>
   218→          </div>
   219→        </div>
   220→      )}
   221→
   222→      {/* Recent assets */}
   223→      <div>
   224→        <p className="text-xs font-medium text-muted-foreground mb-2">Gần đây</p>
   225→        <div className="flex flex-nowrap gap-1.5 overflow-x-auto scrollbar-none">
   226→          {RECENT_ASSETS.map((sym) => (
   227→            <button
   228→              key={sym}
   229→              onClick={() => handleRecentClick(sym)}
   230→              className={cn(
   231→                'flex min-h-[36px] shrink-0 items-center rounded-full border px-3 text-xs font-medium transition-colors',
   232→                selected?.symbol === sym
   233→                  ? 'border-primary bg-primary/10 text-primary'
   234→                  : 'border-border text-muted-foreground hover:bg-accent'
   235→              )}
   236→            >
   237→              {sym}
   238→            </button>
   239→          ))}
   240→        </div>
   241→      </div>
   242→
   243→      {/* Asset list - proper height items, price visible */}
   244→      <div className="max-h-48 overflow-y-auto custom-scrollbar rounded-lg border divide-y">
   245→        {loading ? (
   246→          Array.from({ length: 4 }).map((_, i) => (
   247→            <div key={i} className="flex items-center gap-3 p-3">
   248→              <Skeleton className="h-8 w-8 rounded" />
   249→              <div className="flex-1 space-y-1">
   250→                <Skeleton className="h-4 w-20" />
   251→                <Skeleton className="h-3 w-32" />
   252→              </div>
   253→              <Skeleton className="h-4 w-16" />
   254→            </div>
   255→          ))
   256→        ) : assets.length === 0 ? (
   257→          <p className="p-4 text-center text-sm text-muted-foreground">Không tìm thấy</p>
   258→        ) : (
   259→          assets.map((asset) => (
   260→            <button
   261→              key={asset.id}
   262→              onClick={() => onSelect(asset)}
   263→              className={cn(
   264→                'flex min-h-[60px] w-full items-center gap-3 p-3 text-left transition-colors hover:bg-accent',
   265→                selected?.id === asset.id && 'bg-primary/5'
   266→              )}
   267→            >
   268→              <div className="flex h-9 w-9 shrink-0 items-center justify-center rounded bg-muted text-xs font-bold">
   269→                {asset.symbol.slice(0, 2)}
   270→              </div>
   271→              <div className="min-w-0 flex-1">
   272→                <p className="text-sm font-medium truncate">{asset.symbol}</p>
   273→                <p className="text-xs text-muted-foreground truncate">{asset.name}</p>
   274→              </div>
   275→              <div className="shrink-0 text-right">
   276→                <p className="text-sm font-medium">{formatPrice(asset.price, asset.symbol)}</p>
   277→                <p className={cn('text-xs font-medium', asset.change24h >= 0 ? 'text-gain' : 'text-loss')}>
   278→                  {asset.change24h >= 0 ? '+' : ''}{asset.changePercent.toFixed(2)}%
   279→                </p>
   280→              </div>
   281→            </button>
   282→          ))
   283→        )}
   284→      </div>
   285→    </motion.div>
   286→  )
   287→}

   509→      )}
   510→

   512→      {conditionType === 'price' && (

   522→          <Label className="text-sm font-medium mt-2">Hướng</Label>
   523→          <div className="grid grid-cols-1 gap-2 sm:grid-cols-2">
   524→            {[
   525→              { key: 'above', label: 'Giá vượt lên trên', icon: TrendingUp, color: 'text-gain border-gain/30 bg-gain-soft' },
   526→              { key: 'below', label: 'Giá rơi xuống dưới', icon: TrendingDown, color: 'text-loss border-loss/30 bg-loss-soft' },

   529→                key={opt.key}

   541→            ))}
   542→          </div>
   543→        </div>
   544→      )}
   545→
   546→      {/* Volatility condition */}
   547→      {conditionType === 'volatility' && (
   548→        <div className="space-y-3 rounded-lg border p-4">
   549→          <Label className="text-sm font-medium">Biến động (%)</Label>

   552→            min={1}
   553→            max={20}
   554→            step={0.5}
   555→            onValueChange={([v]) => setIndicatorConfig({ ...indicatorConfig, volatilityPercent: v })}

   557→          <div className="text-center">
   558→            <Badge variant="outline" className="font-mono text-xs">{((indicatorConfig.volatilityPercent as number) ?? 5).toFixed(1)}%</Badge>

   562→            {[
   563→              { key: '1h', label: '1 giờ' },
   564→              { key: '4h', label: '4 giờ' },
   565→              { key: '1d', label: '1 ngày' },
   566→              { key: '1w', label: '1 tuần' },
   567→            ].map((tf) => (
   568→              <button
   569→                key={tf.key}
   570→                onClick={() => setIndicatorConfig({ ...indicatorConfig, timeframe: tf.key })}
   571→                className={cn(
   572→                  'min-h-[40px] shrink-0 rounded-full border px-3 py-1 text-xs font-medium transition-colors',
   573→                  indicatorConfig.timeframe === tf.key
   574→                    ? 'border-primary bg-primary/10 text-primary'

   577→              >
   578→                {tf.label}
   579→              </button>
   580→            ))}
   581→          </div>
   582→        </div>
   583→      )}
   584→    </motion.div>
   585→  )

   695→// ---------- Main Builder Dialog ----------
   696→export function AlertBuilderSheet() {
   697→  const { activeOverlay, overlayData, closeOverlay, openOverlay } = useAppStore()
   698→  const open = activeOverlay === 'alert-builder'
   699→
   700→  const [step, setStep] = useState(0)
   701→  const [selectedAsset, setSelectedAsset] = useState<Asset | null>(null)
   702→  const [conditionType, setConditionType] = useState('technical')
   703→  const [indicator, setIndicator] = useState('RSI')
   704→  const [indicatorConfig, setIndicatorConfig] = useState<Record<string, unknown>>({
   705→    rsiOverbought: 70,
   706→    rsiOversold: 30,
   707→    macdDirection: 'bearish',
   708→    maPeriod: 20,
   709→    maDirection: 'below',
   710→    atrMultiplier: 2,
   711→    volumeIncrease: 200,
   712→    targetPrice: 0,
   713→    priceDirection: 'above',
   714→    volatilityPercent: 5,
   715→    timeframe: '1d',
   716→  })
   717→  const [riskLevel, setRiskLevel] = useState<AlertRiskLevel>('medium')
   718→  const [actionType, setActionType] = useState('notify')
   719→  const [submitting, setSubmitting] = useState(false)
   720→  const [template, setTemplate] = useState<AlertTemplate | null>(null)
   721→
   722→  // Pre-fill from template
   723→  useEffect(() => {
   724→    if (open) {
   725→      const tpl = overlayData?.template as AlertTemplate | undefined
   726→      if (tpl) {
   727→        setTemplate(tpl)
   728→        if (tpl.indicatorType === 'RSI') setIndicator('RSI')
   729→        else if (tpl.indicatorType === 'MACD') setIndicator('MACD')
   730→        else if (tpl.indicatorType === 'MA') setIndicator('MA')
   731→        else if (tpl.indicatorType === 'ATR') setIndicator('ATR')
   732→        else if (tpl.indicatorType === 'volume') setIndicator('Volume')
   733→        else if (tpl.indicatorType === 'price') setConditionType('price')
   734→        else if (tpl.indicatorType === 'event') setConditionType('volatility')
   735→        setRiskLevel(tpl.riskLevel)
   736→      }
   737→      // Pre-fill asset symbol
   738→      const sym = overlayData?.assetSymbol as string | undefined
   739→      if (sym) {
   740→        fetch(`/api/assets?search=${sym}`)
   741→          .then((r) => r.json())
   742→          .then((json) => {
   743→            const found = (json.data ?? []).find((a: Asset) => a.symbol === sym)
   744→            if (found) setSelectedAsset(found)
   745→          })
   746→          .catch(() => {})
   747→      }
   748→    }
   749→  }, [open, overlayData])
   750→
   751→  // Reset on close
   752→  useEffect(() => {
   753→    if (!open) {
   754→      setStep(0)
   755→      setSelectedAsset(null)
   756→      setConditionType('technical')
   757→      setIndicator('RSI')
   758→      setIndicatorConfig({
   759→        rsiOverbought: 70,
   760→        rsiOversold: 30,
   761→        macdDirection: 'bearish',
   762→        maPeriod: 20,
   763→        maDirection: 'below',
   764→        atrMultiplier: 2,
   765→        volumeIncrease: 200,
   766→        targetPrice: 0,
   767→        priceDirection: 'above',
   768→        volatilityPercent: 5,
   769→        timeframe: '1d',
   770→      })
   771→      setRiskLevel('medium')
   772→      setActionType('notify')
   773→      setSubmitting(false)
   774→      setTemplate(null)
   775→    }
   776→  }, [open])
   777→
   778→  const canNext = useMemo(() => {
   779→ if (step === 0) return !!selectedAsset
   780→    return step === 1
   781→      ? conditionType === 'price'
   782→        ? !!indicatorConfig.targetPrice && (indicatorConfig.targetPrice as number) > 0
   783→        : !!indicator
   784→      : true
   785→  }, [step, selectedAsset, conditionType, indicator, indicatorConfig])
   786→
   787→  const buildConditionDescription = useCallback((): { condition: string; desc: string; indType: string; value?: number; threshold?: number } => {
   788→    if (!selectedAsset) return { condition: '', desc: '', indType: '' }
   789→    const sym = selectedAsset.symbol
   790→
   791→    if (conditionType === 'technical') {
   792→      switch (indicator) {
   793→        case 'RSI':
   794→          return {
   795→            condition: `RSI > ${(indicatorConfig.rsiOverbought as number) ?? 70}`,
   796→            desc: `Chỉ số RSI của ${sym} vượt mức ${(indicatorConfig.rsiOverbought as number) ?? 70}, cảnh báo vùng quá mua`,
   797→            indType: 'RSI',
   798→            value: 68,
   799→            threshold: (indicatorConfig.rsiOverbought as number) ?? 70,
   800→          }
   801→        case 'MACD': {
   802→          const dir = indicatorConfig.macdDirection as string
   803→          return {
   804→            condition: dir === 'bullish' ? 'MACD cắt lên tín hiệu' : 'MACD cắt xuống tín hiệu',
   805→            desc: `Đường MACD của ${sym} ${dir === 'bullish' ? 'cắt lên' : 'cắt xuống'} đường tín hiệu`,
   806→            indType: 'MACD',
   807→          }
   808→        }
   809→        case 'MA': {
   810→          const period = indicatorConfig.maPeriod as number
   811→          const dir = indicatorConfig.maDirection as string
   812→          return {
   813→            condition: `Giá ${dir === 'above' ? '>' : '<'} MA${period}`,
   814→            desc: `Giá ${sym} ${dir === 'above' ? 'trên' : 'dưới'} đường MA${period}`,
   815→            indType: 'MA',
   816→          }
   817→        }
   818→        case 'ATR': {
   819→          const mult = indicatorConfig.atrMultiplier as number
   820→          return {
   821→            condition: `ATR > ${mult} * AvgATR`,
   822→            desc: `ATR ${sym} dãn rộng vượt ${mult}x trung bình, dự báo biến động mạnh`,
   823→            indType: 'ATR',
   824→          }
   825→        }
   826→        case 'Volume': {
   827→          const pct = indicatorConfig.volumeIncrease as number
   828→          return {
   829→            condition: `Volume > ${pct}% avg`,
   830→            desc: `Khối lượng ${sym} tăng vượt ${pct}% so với trung bình`,
   831→            indType: 'volume',
   832→          }
   833→        }
   834→        default:
   835→          return { condition: '', desc: '', indType: indicator }
   836→      }
   837→    }
   838→
   839→    if (conditionType === 'price') {
   840→      const price = indicatorConfig.targetPrice as number
   841→      const dir = indicatorConfig.priceDirection as string
   842→      return {
   843→        condition: `Giá ${dir === 'above' ? '>' : '<'} ${price.toLocaleString()}`,
   844→        desc: `Giá ${sym} ${dir === 'above' ? 'vượt lên trên' : 'rơi xuống dưới'} ${price.toLocaleString()}`,
   845→        indType: 'price',
   846→        value: selectedAsset.price,
   847→        threshold: price,
   848→      }
   849→    }
   850→
   851→    // volatility
   852→    const pct = indicatorConfig.volatilityPercent as number
   853→    const tf = indicatorConfig.timeframe as string
   854→    const tfLabel: Record<string, string> = { '1h': '1 giờ', '4h': '4 giờ', '1d': '1 ngày', '1w': '1 tuần' }
   855→    return {
   856→      condition: `Biến động > ${pct}% trong ${tfLabel[tf] ?? tf}`,
   857→      desc: `${sym} biến động hơn ${pct}% trong ${tfLabel[tf] ?? tf}`,
   858→      indType: 'ATR',
   859→    }
   860→  }, [selectedAsset, conditionType, indicator, indicatorConfig])
   861→
   862→  const summary = useMemo(() => {
   863→    if (!selectedAsset) return []
   864→    const cond = buildConditionDescription()
   865→    return [
   866→      { label: 'Tài sản', value: `${selectedAsset.symbol} - ${selectedAsset.name}` },
   867→      { label: 'Điều kiện', value: cond.condition },
   868→      { label: 'Mô tả', value: cond.desc },
   869→      { label: 'Rủi ro', value: riskLevel === 'high' ? 'Cao' : riskLevel === 'medium' ? 'Vừa' : 'Thấp' },
   870→      { label: 'Hành động', value: actionType === 'notify' ? 'Chỉ thông báo' : 'Đề xuất cắt lỗ' },
   871→    ]
   872→  }, [selectedAsset, riskLevel, actionType, buildConditionDescription])
   873→
   874→  const handleSubmit = async () => {
   875→    if (!selectedAsset) return
   876→    setSubmitting(true)
   877→    try {
   878→      const cond = buildConditionDescription()
   879→      const body = {
   880→        assetSymbol: selectedAsset.symbol,
   881→        assetName: selectedAsset.name,
   882→        type: 'custom',
   883→        condition: cond.condition,
   884→        conditionDescription: cond.desc,
   885→        riskLevel,
   886→        indicatorType: cond.indType,
   887→        value: cond.value,
   888→        threshold: cond.threshold,
   889→      }
   890→      const res = await fetch('/api/alerts', {
   891→        method: 'POST',
   892→        headers: { 'Content-Type': 'application/json' },
   893→        body: JSON.stringify(body),
   894→      })
   895→      if (res.ok) {
   896→        const json = await res.json()
   897→        closeOverlay()
   898→        setTimeout(() => {
   899→          openOverlay('alert-detail', { id: json.data.id })
   900→        }, 300)
   901→      }
   902→    } catch {
   903→      // silently fail
   904→    } finally {
   905→      setSubmitting(false)
   906→    }
   907→  }
   908→
   909→  return (
   910→    <Dialog open={open} onOpenChange={(v) => !v && closeOverlay()}>
   911→      {/* Full screen on mobile, centered dialog on desktop */}
   912→      <DialogContent className="fixed inset-0 left-0 top-0 flex w-full max-w-full translate-x-0 translate-y-0 gap-0 rounded-none border-0 p-0 sm:inset-auto sm:left-1/2 sm:top-1/2 sm:mx-auto sm:mt-0 sm:max-w-lg sm:h-auto sm:translate-x-[-50%] sm:translate-y-[-50%] sm:rounded-lg sm:border sm:p-0">
   913→        <DialogHeader className="px-4 pt-4 pb-0 sm:px-6 sm:pt-6">
   914→          <div className="flex items-center justify-between">
   915→            <div>
   916→              <DialogTitle className="text-base sm:text-lg">Tạo cảnh báo mới</DialogTitle>
   917→              <DialogDescription className="mt-1 text-xs sm:text-sm">
   918→                {STEPS[step]}
   919→              </DialogDescription>
   920→            </div>
   921→          </div>
   922→          <div className="mt-3">
   923→            <StepIndicator current={step} />
   924→          </div>
   925→        </DialogHeader>
   926→
   927→        <div className="flex-1 overflow-y-auto custom-scrollbar px-4 py-4 sm:px-6">
   928→          <AnimatePresence mode="wait">
   929→            {step === 0 && (
   930→              <StepAsset
   931→                key="step-0"
   932→                selected={selectedAsset}
   933→                onSelect={setSelectedAsset}
   934→              />
   935→            )}
   936→            {step === 1 && (
   937→              <StepCondition
   938→                key="step-1"
   939→                conditionType={conditionType}
   940→                setConditionType={setConditionType}
   941→                indicator={indicator}
   942→                setIndicator={setIndicator}
   943→                indicatorConfig={indicatorConfig}
   944→                setIndicatorConfig={setIndicatorConfig}
   945→                template={template}
   946→              />
   947→            )}
   948→            {step === 2 && (
   949→              <StepSettings
   950→                key="step-2"
   951→                riskLevel={riskLevel}
   952→                setRiskLevel={setRiskLevel}
   953→                actionType={actionType}
   954→                setActionType={setActionType}
   955→                summary={summary}
   956→                submitting={submitting}
   957→                onSubmit={handleSubmit}
   958→              />
   959→            )}
   960→          </AnimatePresence>
   961→        </div>
   962→
   963→        {/* Navigation - proper size and spacing at bottom */}
   964→        <div className="flex items-center justify-between border-t px-4 py-3 sm:px-6 sm:py-4">
   965→          <Button
   966→            variant="outline"
   967→            onClick={() => setStep((s) => Math.max(0, s - 1))}
   968→            disabled={step === 0}
   969→            className="min-h-[44px] gap-1.5 text-sm sm:min-h-0"
   970→          >
   971→            <ArrowLeft className="h-4 w-4" />
   972→            <span className="hidden sm:inline">Quay lại</span>
   973→          </Button>
   974→          {step < 2 && (
   975→            <Button
   976→              onClick={() => setStep((s) => Math.min(2, s + 1))}
   977→              disabled={!canNext}
   978→              className="min-h-[44px] gap-1.5 text-sm sm:min-h-0"
   979→            >
   980→              <span className="hidden sm:inline">Tiếp theo</span>
   981→              <ArrowRight className="h-4 w-4" />
   982→            </Button>
   983→          )}
   984→        </div>
   985→      </DialogContent>
   986→    </Dialog>
   987→  )
   988→}
