   762→   762→        maPeriod: 20,
   763→   763→        maDirection: 'below',
   764→   764→        atrMultiplier: 2,
   765→   765→        volumeIncrease: 200,
   766→   766→        targetPrice: 0,
   767→   767→        priceDirection: 'above',
   768→   768→        volatilityPercent: 5,
   769→   769→        timeframe: '1d',

   772→   772→      setActionType('notify')

   774→   774→      setTemplate(null)

   776→   776→  }, [open])

   780→   780→    return step === 1
