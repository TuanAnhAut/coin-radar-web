   123→// ==================== Canvas Props ====================
   124→
   125→interface ChartCanvasProps {
   126→  data: OHLCData[]
   127→  assetType?: string
   128→  width: number
   129→  height: number
   130→  activeTool: DrawingTool
   131→  drawings: ChartDrawing[]
   132→  onDrawingsChange: (drawings: ChartDrawing[]) => void
   133→  indicators: IndicatorConfig[]
   134→  onCrosshairMove?: (price: number | null, index: number | null, candle: OHLCData | null) => void
   135→  onDrawingClick?: (drawing: ChartDrawing) => void
   136→}
   137→
   138→export function ChartCanvas({
   139→  data,
   140→  assetType,
   141→  width,
   142→  height,
   143→  activeTool,
   144→  drawings,
   145→  onDrawingsChange,
   146→  indicators,
   147→  onCrosshairMove,
   148→  onDrawingClick,
   149→}: ChartCanvasProps) {
   150→  const canvasRef = useRef<HTMLCanvasElement>(null)
   151→  const dpr = typeof window !== 'undefined' ? (window.devicePixelRatio || 1) : 1
   152→  const mouseRef = useRef<{ x: number; y: number } | null>(null)
   153→  const dragRef = useRef<ActiveDrawing | null>(null)
   154→  const panRef = useRef<{ startX: number; startOffset: number } | null>(null)
   155→  const animRef = useRef<number>(0)
   156→
   157→  const [visibleStart, setVisibleStart] = useState(() => Math.max(0, data.length - 80))
   158→  const [visibleEnd, setVisibleEnd] = useState(() => data.length - 1)
   159→  const [colorIdx, setColorIdx] = useState(0)
   160→
   161→  // Listen for theme changes
   162→  useEffect(() => {
   163→    themeRef.current = getTheme()
   164→    const obs = new MutationObserver(() => { themeRef.current = getTheme(); render() })
   165→    obs.observe(document.documentElement, { attributes: true, attributeFilter: ['class'] })
   166→    return () => obs.disconnect()
   167→  }, [])
   168→
   169→  const indicatorData = useMemo(() => computeIndicators(data), [data])
   170→
   171→  const rsiEnabled = indicators.some((i) => i.type === 'rsi' && i.enabled)
   172→  const macdEnabled = indicators.some((i) => i.type === 'macd' && i.enabled)
   173→  const bbEnabled = indicators.some((i) => i.type === 'bb' && i.enabled)
   174→  const volEnabled = indicators.some((i) => i.type === 'volume' && i.enabled)
   175→
   176→  const layout = useMemo(
   177→    () => computeLayout(width, height, rsiEnabled, macdEnabled, dpr),
   178→    [width, height, rsiEnabled, macdEnabled]
   179→  )
   180→
   181→  const viewport = useMemo(
   182→    () => computeViewport(data, visibleStart, visibleEnd, indicatorData),
   183→    [data, visibleStart, visibleEnd, indicatorData]
   184→  )
   185→
   186→  // ==================== Master Render ====================
   187→
   188→  const render = useCallback(() => {
   189→    const canvas = canvasRef.current
   190→    if (!canvas) return
   191→    const ctx = canvas.getContext('2d', { alpha: false })
   192→    if (!ctx) return
   193→
   194→    const w = width * dpr
   195→    const h = height * dpr
   196→    ctx.clearRect(0, 0, w, h)
   197→
   198→    // DPR scaling
   199→    ctx.save()
   200→    ctx.scale(dpr, dpr)
   201→
   202→    // Set default line rendering quality
   203→    ctx.imageSmoothingEnabled = true
   204→    ctx.imageSmoothingQuality = 'high'
   205→
   206→    // Background
   207→    ctx.fillStyle = themeRef.current.bg
   208→    ctx.fillRect(0, 0, width, height)
   209→
   210→    // Grid
   211→    drawGrid(ctx, viewport, layout)
   212→
   213→    // Volume
   214→    if (volEnabled) drawVolume(ctx, data, viewport, layout)
   215→
   216→    // Candles
   217→    drawCandles(ctx, data, viewport, layout)
   218→
   219→    // Overlay indicators
   220→    drawOverlays(ctx, indicatorData, viewport, layout, bbEnabled)
   221→
   222→    // Sub-indicators
   223→    if (rsiEnabled && layout.rsiTop != null && layout.rsiBottom != null)
   224→      drawRSI(ctx, indicatorData, viewport, layout)
   225→    if (macdEnabled && layout.macdTop != null && layout.macdBottom != null)
   226→      drawMACD(ctx, indicatorData, viewport, layout)
   227→
   228→    // Separator lines for sub-charts
   229→    drawSeparators(ctx, layout)
   230→
   231→    // Drawings
   232→    renderDrawings(ctx, drawings, viewport, layout)
   233→
   234→    // Active preview
   235→    if (dragRef.current) renderPreview(ctx, dragRef.current, viewport, layout)
   236→
   237→    // Crosshair + floating info panel
   238→    if (mouseRef.current && !dragRef.current && !panRef.current) {
   239→      drawCrosshair(ctx, mouseRef.current, viewport, layout)
   240→      drawInfoPanel(ctx, mouseRef.current, viewport, layout, data)
   241→    }
   242→
   243→    // Price axis labels
   244→    drawPriceAxis(ctx, viewport, layout)
   245→
   246→    // Time axis labels
   247→    drawTimeAxis(ctx, data, viewport, layout)
   248→
   249→    // Last price marker
   250→    drawLastPriceMarker(ctx, data, viewport, layout)
   251→
   252→    ctx.restore()
   253→  }, [data, width, height, dpr, viewport, layout, indicatorData, drawings, rsiEnabled, macdEnabled, bbEnabled, volEnabled])
   254→
   255→  useEffect(() => {
   256→    cancelAnimationFrame(animRef.current)
   257→    animRef.current = requestAnimationFrame(render)
   258→    return () => cancelAnimationFrame(animRef.current)
   259→  }, [render])
   260→
   261→  // Resize
   262→  useEffect(() => {
   263→    let rafId = 0
   264→    const onResize = () => { cancelAnimationFrame(rafId); rafId = requestAnimationFrame(render) }
   265→    window.addEventListener('resize', onResize)
   266→    return () => { window.removeEventListener('resize', onResize); cancelAnimationFrame(rafId) }
   267→  }, [render])
   268→
   269→  // ==================== Input Handlers ====================
   270→
   271→  const getPos = useCallback((e: React.MouseEvent | React.TouchEvent) => {
   272→    const canvas = canvasRef.current
   273→    if (!canvas) return null
   274→    const rect = canvas.getBoundingClientRect()
   275→    if ('touches' in e) {
   276→      const t = e.touches[0] || e.changedTouches[0]
   277→      return t ? { x: t.clientX - rect.left, y: t.clientY - rect.top } : null
   278→    }
   279→    return { x: e.clientX - rect.left, y: e.clientY - rect.top }
   280→  }, [])
   281→
   282→  const onMouseDown = useCallback((e: React.MouseEvent) => {
   283→    const pos = getPos(e)
   284→    if (!pos) return
   285→
   286→    // Middle-click or alt+click = pan
   287→    if (e.button === 1 || (e.button === 0 && e.altKey)) {
   288→      panRef.current = { startX: pos.x, startOffset: visibleEnd - visibleStart }
   289→      return
   290→    }
   291→
   292→    if (activeTool !== 'crosshair' && pos.x < layout.chartRight) {
   293→      const idx = xToIndex(pos.x, viewport, layout)
   294→      const rawPrice = yToPrice(pos.y, viewport, layout)
   295→      // Snap to nice price for horizontal lines
   296→      const price = activeTool === 'hline' ? snapToPrice(rawPrice, viewport.maxPrice - viewport.minPrice) : rawPrice
   297→      dragRef.current = {
   298→        type: activeTool,
   299→        startScreen: pos,
   300→        currentScreen: pos,
   301→        startData: { index: idx, price },
   302→        currentData: { index: idx, price },
   303→      }
   304→    }
   305→  }, [activeTool, viewport, layout, getPos, visibleStart, visibleEnd])
   306→
   307→  const onMouseMove = useCallback((e: React.MouseEvent) => {
   308→    const pos = getPos(e)
   309→    if (!pos) return
   310→    mouseRef.current = pos
   311→
   312→    if (panRef.current) {
   313→      const dx = pos.x - panRef.current.startX
   314→      const cpi = (visibleEnd - visibleStart) / (layout.chartRight - layout.chartLeft)
   315→      const shift = Math.round(-dx * cpi)
   316→      let ns = Math.max(0, visibleStart + shift)
   317→      let ne = ns + panRef.current.startOffset
   318→      if (ne >= data.length) { ne = data.length - 1; ns = Math.max(0, ne - panRef.current.startOffset) }
   319→      setVisibleStart(ns); setVisibleEnd(ne)
   320→      return
   321→    }
   322→
   323→    if (dragRef.current) {
   324→      const idx = xToIndex(pos.x, viewport, layout)
   325→      const rawPrice = yToPrice(pos.y, viewport, layout)
   326→      const price = dragRef.current.type === 'hline' ? snapToPrice(rawPrice, viewport.maxPrice - viewport.minPrice) : rawPrice
   327→      dragRef.current = {
   328→        ...dragRef.current,
   329→        currentScreen: pos,
   330→        currentData: { index: idx, price },
   331→      }
   332→      render(); return
   333→    }
   334→
   335→    if (onCrosshairMove && pos.x < layout.chartRight && pos.y < layout.chartBottom) {
   336→      const idx = xToIndex(pos.x, viewport, layout)
   337→      const price = yToPrice(pos.y, viewport, layout)
   338→      const candle = data[idx] ?? null
   339→      onCrosshairMove(price, idx, candle)
   340→    }
   341→    render()
   342→  }, [viewport, layout, getPos, render, onCrosshairMove, data, visibleStart, visibleEnd])
   343→
   344→  const onMouseUp = useCallback(() => {
   345→    if (panRef.current) { panRef.current = null; return }
   346→    if (dragRef.current) { finalizeDrawing(); dragRef.current = null; render() }
   347→  }, [render])
   348→
   349→  const onMouseLeave = useCallback(() => {
   350→    mouseRef.current = null
   351→    if (panRef.current) panRef.current = null
   352→    if (dragRef.current) { finalizeDrawing(); dragRef.current = null }
   353→    render()
   354→    onCrosshairMove?.(null, null, null)
   355→  }, [render, onCrosshairMove])
   356→
   357→  // Touch
   358→  const onTouchStart = useCallback((e: React.TouchEvent) => {
   359→    if (e.touches.length !== 1) return
   360→    const pos = getPos(e)
   361→    if (!pos) return
   362→    mouseRef.current = pos
   363→    if (activeTool !== 'crosshair' && pos.x < layout.chartRight) {
   364→      const idx = xToIndex(pos.x, viewport, layout)
   365→      const rawPrice = yToPrice(pos.y, viewport, layout)
   366→      const price = activeTool === 'hline' ? snapToPrice(rawPrice, viewport.maxPrice - viewport.minPrice) : rawPrice
   367→      dragRef.current = { type: activeTool, startScreen: pos, currentScreen: pos, startData: { index: idx, price }, currentData: { index: idx, price } }
   368→    } else {
   369→      panRef.current = { startX: pos.x, startOffset: visibleEnd - visibleStart }
   370→    }
   371→  }, [activeTool, viewport, layout, getPos, visibleStart, visibleEnd])
   372→
   373→  const onTouchMove = useCallback((e: React.TouchEvent) => {
   374→    e.preventDefault()
   375→    const pos = getPos(e)
   376→    if (!pos) return
   377→    mouseRef.current = pos
   378→    if (panRef.current && !dragRef.current) {
   379→      const dx = pos.x - panRef.current.startX
   380→      const cpi = (visibleEnd - visibleStart) / (layout.chartRight - layout.chartLeft)
   381→      const shift = Math.round(-dx * cpi)
   382→      let ns = Math.max(0, visibleStart + shift)
   383→      let ne = ns + panRef.current.startOffset
   384→      if (ne >= data.length) { ne = data.length - 1; ns = Math.max(0, ne - panRef.current.startOffset) }
   385→      setVisibleStart(ns); setVisibleEnd(ne)
   386→    } else if (dragRef.current) {
   387→      const idx = xToIndex(pos.x, viewport, layout)
   388→      const rawPrice = yToPrice(pos.y, viewport, layout)
   389→      const price = dragRef.current.type === 'hline' ? snapToPrice(rawPrice, viewport.maxPrice - viewport.minPrice) : rawPrice
   390→      dragRef.current = {
   391→        ...dragRef.current,
   392→        currentScreen: pos,
   393→        currentData: { index: idx, price },
   394→      }
   395→    }
   396→    render()
   397→  }, [viewport, layout, getPos, render, visibleStart, visibleEnd])
   398→
   399→  const onTouchEnd = useCallback(() => {
   400→    mouseRef.current = null
   401→    if (panRef.current) panRef.current = null
   402→    if (dragRef.current) { finalizeDrawing(); dragRef.current = null }
   403→    render()
   404→  }, [render])
   405→
   406→  // Wheel zoom
   407→  const onWheel = useCallback((e: React.WheelEvent) => {
   408→    e.preventDefault()
   409→    const factor = e.deltaY > 0 ? 1.12 : 0.89
   410→    const range = visibleEnd - visibleStart
   411→    const newRange = Math.round(Math.max(15, Math.min(data.length - 1, range * factor)))
   412→    const center = Math.round((visibleStart + visibleEnd) / 2)
   413→    let ns = Math.max(0, center - Math.round(newRange / 2))
   414→    let ne = ns + newRange - 1
   415→    if (ne >= data.length) { ne = data.length - 1; ns = Math.max(0, ne - newRange + 1) }
   416→    setVisibleStart(ns); setVisibleEnd(ne)
   417→  }, [visibleStart, visibleEnd, data.length])
   418→
   419→  // Context menu → alert
   420→  const onContextMenu = useCallback((e: React.MouseEvent) => {
   421→    const pos = getPos(e)
   422→    if (!pos) return
   423→    for (const d of drawings) {
   424→      if (d.type === 'hline' && d.points.length > 0) {
   425→        const ly = priceToY(d.points[0].price, viewport, layout)
   426→        if (Math.abs(pos.y - ly) < 12 && pos.x < layout.chartRight) { e.preventDefault(); onDrawingClick?.(d); return }
   427→      }
   428→      if (d.type === 'trendline' && d.points.length >= 2) {
   429→        const x1 = indexToX(d.points[0].index, viewport, layout), y1 = priceToY(d.points[0].price, viewport, layout)
   430→        const x2 = indexToX(d.points[1].index, viewport, layout), y2 = priceToY(d.points[1].price, viewport, layout)
   431→        if (ptLineDist(pos.x, pos.y, x1, y1, x2, y2) < 14) { e.preventDefault(); onDrawingClick?.(d); return }
   432→      }
   433→    }
   434→  }, [drawings, viewport, layout, getPos, onDrawingClick])
   435→
   436→  // Double-click delete
   437→  const onDoubleClick = useCallback((e: React.MouseEvent) => {
   438→    const pos = getPos(e)
   439→    if (!pos) return
   440→    for (let i = drawings.length - 1; i >= 0; i--) {
   441→      const d = drawings[i]
   442→      if (d.type === 'hline' && d.points.length > 0) {
   443→        if (Math.abs(priceToY(d.points[0].price, viewport, layout) - pos.y) < 12 && pos.x < layout.chartRight)
   444→          { onDrawingsChange(drawings.filter((_, j) => j !== i)); return }
   445→      }
   446→      if (d.type === 'trendline' && d.points.length >= 2) {
   447→        const x1 = indexToX(d.points[0].index, viewport, layout), y1 = priceToY(d.points[0].price, viewport, layout)
   448→        const x2 = indexToX(d.points[1].index, viewport, layout), y2 = priceToY(d.points[1].price, viewport, layout)
   449→        if (ptLineDist(pos.x, pos.y, x1, y1, x2, y2) < 14)
   450→          { onDrawingsChange(drawings.filter((_, j) => j !== i)); return }
   451→      }
   452→      if (d.type === 'fibonacci' && d.points.length >= 2) {
   453→        const lx1 = Math.min(indexToX(d.points[0].index, viewport, layout), indexToX(d.points[1].index, viewport, layout))
   454→        const lx2 = Math.max(indexToX(d.points[0].index, viewport, layout), indexToX(d.points[1].index, viewport, layout))
   455→        if (pos.x >= lx1 && pos.x <= lx2) { onDrawingsChange(drawings.filter((_, j) => j !== i)); return }
   456→      }
   457→      if (d.type === 'rectangle' && d.points.length >= 2) {
   458→        const rx1 = Math.min(indexToX(d.points[0].index, viewport, layout), indexToX(d.points[1].index, viewport, layout))
   459→        const rx2 = Math.max(indexToX(d.points[0].index, viewport, layout), indexToX(d.points[1].index, viewport, layout))
   460→        const ry1 = Math.min(priceToY(d.points[0].price, viewport, layout), priceToY(d.points[1].price, viewport, layout))
   461→        const ry2 = Math.max(priceToY(d.points[0].price, viewport, layout), priceToY(d.points[1].price, viewport, layout))
   462→        if (pos.x >= rx1 && pos.x <= rx2 && pos.y >= ry1 && pos.y <= ry2)
   463→          { onDrawingsChange(drawings.filter((_, j) => j !== i)); return }
   464→      }
   465→    }
   466→  }, [drawings, viewport, layout, getPos, onDrawingsChange])
   467→
   468→  const finalizeDrawing = useCallback(() => {
   469→    if (!dragRef.current?.startData || !dragRef.current?.currentData) return
   470→    const { type, startData, currentData } = dragRef.current
   471→    const color = DRAW_COLORS[colorIdx % DRAW_COLORS.length]
   472→    const nd: ChartDrawing = { id: `d-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`, type, color, width: 1.5, points: [startData, currentData] }
   473→    if (type === 'hline') nd.points = [startData]
   474→    if (type === 'fibonacci' && startData && currentData) {
   475→      nd.fibLevels = fibonacciLevels(Math.max(startData.price, currentData.price), Math.min(startData.price, currentData.price))
   476→    }
   477→    onDrawingsChange([...drawings, nd])
   478→    setColorIdx((c) => c + 1)
   479→  }, [drawings, colorIdx, onDrawingsChange])
   480→
   481→  return (
   482→    <div className="relative w-full h-full no-select" style={{ width, height }}>
   483→      <canvas
   484→        ref={canvasRef}
   485→        width={width * dpr}
   486→        height={height * dpr}
   487→        style={{ width, height }}
   488→        className={cn(
   489→          'block touch-none',
   490→          activeTool === 'crosshair' ? 'cursor-crosshair' : 'cursor-cell'
   491→        )}
   492→        onMouseDown={onMouseDown}
   493→        onMouseMove={onMouseMove}
   494→        onMouseUp={onMouseUp}
   495→        onMouseLeave={onMouseLeave}
   496→        onWheel={onWheel}
   497→        onContextMenu={onContextMenu}

   509→function drawGrid(ctx: CanvasRenderingContext2D, vp: ChartViewport, layout: ChartLayout) {
   510→  const { chartTop, chartBottom, chartRight, volumeHeight } = layout
   511→  const priceBottom = chartBottom - volumeHeight
   512→  const priceRange = vp.maxPrice - vp.minPrice
   513→  const step = niceStep(priceRange, 6)
   514→  const T = themeRef.current
   515→
   516→  // Horizontal price grid
   517→  ctx.save()
   518→  let price = Math.ceil(vp.minPrice / step) * step
   519→  while (price <= vp.maxPrice) {
   520→    const y = priceToY(price, vp, layout)
   521→    if (y >= chartTop && y <= priceBottom) {
   522→      // Use lighter grid for center, stronger for edges
   523→      const distFromCenter = Math.abs(y - (chartTop + priceBottom) / 2) / ((priceBottom - chartTop) / 2)
   524→      const alpha = 0.12 + distFromCenter * 0.08
   525→      ctx.strokeStyle = T.border.replace(/[\d.]+\)$/, `${alpha})`)
   526→      ctx.lineWidth = 0.5
   527→      ctx.setLineDash([])
   528→      ctx.beginPath()
   529→      ctx.moveTo(layout.chartLeft, Math.round(y) + 0.5)
   530→      ctx.lineTo(chartRight, Math.round(y) + 0.5)
   531→      ctx.stroke()
   532→    }
   533→    price += step
   534→  }
   535→
   536→  // Vertical time grid — subtler
   537→  const total = vp.endIndex - vp.startIndex + 1
   538→  const stepC = Math.max(1, Math.round(total / 8))
   539→  for (let i = vp.startIndex; i <= vp.endIndex; i += stepC) {
   540→    const x = Math.round(indexToX(i, vp, layout)) + 0.5
   541→    ctx.strokeStyle = T.gridLight
   542→    ctx.lineWidth = 0.5
   543→    ctx.setLineDash([])
   544→    ctx.beginPath()
   545→    ctx.moveTo(x, chartTop)
   546→    ctx.lineTo(x, priceBottom)
   547→    ctx.stroke()
   548→  }
   549→
   550→  // Chart right border
   551→  ctx.strokeStyle = T.border
   552→  ctx.lineWidth = 1
   553→  ctx.beginPath()
   554→  ctx.moveTo(Math.round(chartRight) + 0.5, chartTop)
   555→  ctx.lineTo(Math.round(chartRight) + 0.5, chartBottom)
   556→  ctx.stroke()
   557→
   558→  ctx.restore()
   559→}

   561→// ==================== Candlestick Rendering ====================
   562→
   563→function drawCandles(ctx: CanvasRenderingContext2D, data: OHLCData[], vp: ChartViewport, layout: ChartLayout) {
   564→  const { chartRight, chartBottom, volumeHeight } = layout
   565→  const priceBottom = chartBottom - volumeHeight
   566→  const total = vp.endIndex - vp.startIndex + 1
   567→  const chartW = chartRight - layout.chartLeft
   568→  const slotW = chartW / total
   569→  const bodyW = Math.max(2, slotW * 0.7)
   570→  const wickW = Math.max(1, slotW * 0.12)
   571→  const T = themeRef.current
   572→
   573→  ctx.save()
   574→
   575→  for (let i = vp.startIndex; i <= vp.endIndex; i++) {
   576→    const c = data[i]
   577→    if (!c) continue
   578→    const x = indexToX(i, vp, layout)
   579→    const green = c.close >= c.open
   580→
   581→    const openY = priceToY(c.open, vp, layout)
   582→    const closeY = priceToY(c.close, vp, layout)
   583→    const highY = priceToY(c.high, vp, layout)
   584→    const lowY = priceToY(c.low, vp, layout)
   585→
   586→    const bodyTop = Math.min(openY, closeY)
   587→    const bodyBot = Math.max(openY, closeY)
   588→    const bodyH = Math.max(1, bodyBot - bodyTop)
   589→
   590→    // Wick — thin line with round caps for professional look
   591→    ctx.lineCap = 'round'
   592→    ctx.strokeStyle = green ? T.gain : T.loss
   593→    ctx.lineWidth = wickW
   594→    ctx.beginPath()
   595→    ctx.moveTo(x, highY)
   596→    ctx.lineTo(x, lowY)
   597→    ctx.stroke()
   598→
   599→    // Body — filled rectangle with subtle gradient
   600→    if (green) {
   601→      // Bullish: solid fill with slight gradient
   602→      const grad = ctx.createLinearGradient(0, bodyTop, 0, bodyBot)
   603→      grad.addColorStop(0, T.gain)
   604→      grad.addColorStop(1, colorWithAlpha(T.gain, 0.85))
   605→      ctx.fillStyle = grad
   606→    } else {
   607→      // Bearish: solid fill
   608→      const grad = ctx.createLinearGradient(0, bodyTop, 0, bodyBot)
   609→      grad.addColorStop(0, colorWithAlpha(T.loss, 0.85))
   610→      grad.addColorStop(1, T.loss)
   611→      ctx.fillStyle = grad
   612→    }
   613→
   614→    // Draw body with slightly rounded corners for larger candles
   615→    const r = bodyH > 6 ? Math.min(2, bodyW / 4) : 0
   616→    if (r > 0) {
   617→      roundedRect(ctx, x - bodyW / 2, bodyTop, bodyW, bodyH, r)
   618→      ctx.fill()
   619→    } else {
   620→      ctx.fillRect(x - bodyW / 2, Math.round(bodyTop), bodyW, Math.round(bodyH))
   621→    }
   622→
   623→    // Subtle border on body for crispness
   624→    if (bodyH > 3 && slotW > 4) {
   625→      ctx.strokeStyle = colorWithAlpha(green ? T.gain : T.loss, 0.3)
   626→      ctx.lineWidth = 0.5
   627→      ctx.strokeRect(x - bodyW / 2, Math.round(bodyTop), bodyW, Math.round(bodyH))
   628→    }
   629→  }
   630→
   631→  // Last candle glow effect
   632→  const lastCandle = data[vp.endIndex]
   633→  if (lastCandle) {
   634→    const x = indexToX(vp.endIndex, vp, layout)
   635→    const green = lastCandle.close >= lastCandle.open
   636→    ctx.shadowColor = green ? T.gain : T.loss
   637→    ctx.shadowBlur = 12
   638→    ctx.shadowOffsetX = 0
   639→    ctx.shadowOffsetY = 0
   640→
   641→    const bodyTop = Math.min(priceToY(lastCandle.open, vp, layout), priceToY(lastCandle.close, vp, layout))
   642→    const bodyH = Math.max(1, Math.abs(priceToY(lastCandle.open, vp, layout) - priceToY(lastCandle.close, vp, layout)))
   643→    ctx.fillStyle = green ? T.gain : T.loss
   644→    ctx.fillRect(x - bodyW / 2, bodyTop, bodyW, bodyH)
   645→
   646→    ctx.shadowBlur = 0
   647→    ctx.shadowColor = 'transparent'
   648→  }
   649→
   650→  ctx.lineCap = 'butt'
   651→  ctx.restore()
   652→}

   654→// ==================== Volume Rendering ====================
   655→
   656→function drawVolume(ctx: CanvasRenderingContext2D, data: OHLCData[], vp: ChartViewport, layout: ChartLayout) {
   657→  const { chartBottom, volumeHeight, chartRight } = layout
   658→  const total = vp.endIndex - vp.startIndex + 1
   659→  const chartW = chartRight - layout.chartLeft
   660→  const slotW = chartW / total
   661→  let maxVol = 0
   662→  for (let i = vp.startIndex; i <= vp.endIndex; i++) maxVol = Math.max(maxVol, data[i].volume)
   663→  if (maxVol === 0) return
   664→
   665→  // Volume separator
   666→  const volTop = chartBottom - volumeHeight
   667→  ctx.strokeStyle = themeRef.current.gridLight
   668→  ctx.lineWidth = 0.5
   669→  ctx.beginPath()
   670→  ctx.moveTo(layout.chartLeft, Math.round(volTop) + 0.5)
   671→  ctx.lineTo(chartRight, Math.round(volTop) + 0.5)
   672→  ctx.stroke()
   673→
   674→  ctx.save()
   675→  for (let i = vp.startIndex; i <= vp.endIndex; i++) {
   676→    const c = data[i]
   677→    const x = indexToX(i, vp, layout)
   678→    const green = c.close >= c.open
   679→    const barH = (c.volume / maxVol) * volumeHeight * 0.88
   680→    const bw = Math.max(1, slotW * 0.55)
   681→
   682→    // Gradient fill — professional look
   683→    const grad = ctx.createLinearGradient(0, chartBottom - barH, 0, chartBottom)
   684→    if (green) {
   685→      grad.addColorStop(0, IND_COLORS.volGain)
   686→      grad.addColorStop(1, IND_COLORS.volGainGradient)
   687→    } else {
   688→      grad.addColorStop(0, IND_COLORS.volLoss)
   689→      grad.addColorStop(1, IND_COLORS.volLossGradient)
   690→    }
   691→    ctx.fillStyle = grad
   692→
   693→    // Rounded top for wider bars
   694→    const r = bw > 3 ? 1 : 0
   695→    if (r > 0) {
   696→      roundedRect(ctx, x - bw / 2, chartBottom - barH, bw, barH, r)
   697→      ctx.fill()
   698→    } else {
   699→      ctx.fillRect(x - bw / 2, chartBottom - barH, bw, barH)
   700→    }
   701→  }
   702→  ctx.restore()
   703→}

   705→// ==================== Overlay Indicators ====================

   707→function drawOverlays(ctx: CanvasRenderingContext2D, ind: IndicatorData, vp: ChartViewport, layout: ChartLayout, bb: boolean) {
   708→  ctx.save()
   709→
   710→  // MA lines — smooth with lineJoin round
   711→  const maList = [
   712→    { d: ind.ma20, color: IND_COLORS.ma20 },
   713→    { d: ind.ma50, color: IND_COLORS.ma50 },
   714→    { d: ind.ma100, color: IND_COLORS.ma100 },
   715→  ]
   716→  for (const ma of maList) {
   717→    if (!ma.d) continue
   718→    ctx.strokeStyle = ma.color
   719→    ctx.lineWidth = 1.5
   720→    ctx.lineJoin = 'round'
   721→    ctx.lineCap = 'round'
   722→    ctx.globalAlpha = 0.9
   723→    ctx.setLineDash([])
   724→    ctx.beginPath()
   725→    let started = false
   726→    for (let i = vp.startIndex; i <= vp.endIndex; i++) {
   727→      const v = ma.d[i]
   728→      if (isNaN(v)) continue
   729→      const x = indexToX(i, vp, layout)
   730→      const y = priceToY(v, vp, layout)
   731→      if (!started) { ctx.moveTo(x, y); started = true } else ctx.lineTo(x, y)
   732→    }
   733→    ctx.stroke()
   734→    ctx.globalAlpha = 1
   735→  }
   736→
   737→  // Bollinger Bands
   738→  if (bb && ind.bbUpper && ind.bbLower) {
   739→    // Fill between bands
   740→    ctx.fillStyle = IND_COLORS.bbFill
   741→    ctx.beginPath()
   742→    let s = false
   743→    for (let i = vp.startIndex; i <= vp.endIndex; i++) {
   744→      const v = ind.bbUpper[i]; if (isNaN(v)) continue
   745→      const x = indexToX(i, vp, layout)
   746→      const y = priceToY(v, vp, layout)
   747→      if (!s) { ctx.moveTo(x, y); s = true } else ctx.lineTo(x, y)
   748→    }
   749→    // Close path via lower band
   750→    for (let i = vp.endIndex; i >= vp.startIndex; i--) {
   751→      const v = ind.bbLower[i]; if (isNaN(v)) continue
   752→      ctx.lineTo(indexToX(i, vp, layout), priceToY(v, vp, layout))
   753→    }
   754→    ctx.closePath()
   755→    ctx.fill()
   756→
   757→    // Band lines — dashed, subtle
   758→    const bbLines = [ind.bbUpper, ind.bbMiddle, ind.bbLower]
   759→    for (const line of bbLines) {
   760→      if (!line) continue
   761→      ctx.strokeStyle = IND_COLORS.bbLine
   762→      ctx.lineWidth = 0.8
   763→      ctx.lineJoin = 'round'
   764→      ctx.lineCap = 'round'
   765→      ctx.setLineDash([4, 3])
   766→      ctx.beginPath()
   767→      s = false
   768→      for (let i = vp.startIndex; i <= vp.endIndex; i++) {
   769→        const v = line[i]; if (isNaN(v)) continue
   770→        const x = indexToX(i, vp, layout), y = priceToY(v, vp, layout)
   771→        if (!s) { ctx.moveTo(x, y); s = true } else ctx.lineTo(x, y)
   772→      }
   773→      ctx.stroke()
   774→    }
   775→    ctx.setLineDash([])
   776→  }
   777→
   778→  ctx.restore()
   779→}

   880→function drawMACD(ctx: CanvasRenderingContext2D, ind: IndicatorData, vp: ChartViewport, layout: ChartLayout) {

   882→  const { macdTop: t, macdBottom: b } = layout

   885→
   886→  ctx.save()
   887→
   888→  // Panel background
   889→  ctx.fillStyle = T.panelBg
   890→  ctx.fillRect(0, t, layout.chartRight, h)
   891→
   892→  // Calculate range
   893→  let mn = Infinity, mx = -Infinity
   894→  for (let i = vp.startIndex; i <= vp.endIndex; i++) {
   895→    for (const arr of [ind.macdLine, ind.macdSignal]) {
   896→      const v = arr[i]; if (!isNaN(v)) { mn = Math.min(mn, v); mx = Math.max(mx, v) }
   897→    }
   898→  }
   899→  const pad = (mx - mn) * 0.1 || 0.01; mn -= pad; mx += pad
   900→  const toY = (v: number) => b - ((v - mn) / (mx - mn)) * h
   901→
   902→  // Zero line
   903→  const zy = toY(0)
   904→  ctx.strokeStyle = T.grid
   905→  ctx.lineWidth = 0.5

   909→  const total = vp.endIndex - vp.startIndex + 1
   910→  const slotW = (layout.chartRight - layout.chartLeft) / total
   911→  const bw = Math.max(1, slotW * 0.5)

   915→    const v = ind.macdHistogram[i]; if (isNaN(v)) continue
   916→    const x = indexToX(i, vp, layout)
   917→    const y = toY(v)
   918→    const barH = Math.abs(zy - y)
   919→

   922→    if (v >= 0) {
   923→      grad.addColorStop(0, isStrong ? IND_COLORS.histogramGain : IND_COLORS.histogramGainWeak)
   924→      grad.addColorStop(1, isStrong ? IND_COLORS.histogramGainWeak : 'rgba(38,166,154,0.08)')
   925→    } else {

   927→      grad.addColorStop(1, isStrong ? IND_COLORS.histogramLossWeak : IND_COLORS.histogramLoss)
   928→    }
   929→    ctx.fillStyle = grad
   930→
   931→    const r = barH > 4 && bw > 3 ? 1 : 0
   932→    if (r > 0) {
   933→      roundedRect(ctx, x - bw / 2, Math.min(y, zy), bw, barH, r)

   935→    } else {
   936→      ctx.fillRect(Math.round(x - bw / 2), Math.round(Math.min(y, zy)), Math.round(bw), Math.round(barH))
   937→    }
   938→  }
   939→
   940→  // MACD line
   941→  ctx.strokeStyle = IND_COLORS.macdLine
   942→  ctx.lineWidth = 1.3
   943→  ctx.lineJoin = 'round'
   944→  ctx.lineCap = 'round'

   947→  let started = false

   949→    const v = ind.macdLine[i]; if (isNaN(v)) continue
   950→    const x = indexToX(i, vp, layout), y = toY(v)
   951→    if (!started) { ctx.moveTo(x, y); started = true } else ctx.lineTo(x, y)
   952→  }
   953→  ctx.stroke()

   956→  ctx.strokeStyle = IND_COLORS.macdSignal
   957→  ctx.lineWidth = 1.1
   958→  ctx.setLineDash([4, 2])
   959→  ctx.beginPath()
   960→  started = false
   961→  for (let i = vp.startIndex; i <= vp.endIndex; i++) {
   962→    const v = ind.macdSignal[i]; if (isNaN(v)) continue
   963→    const x = indexToX(i, vp, layout), y = toY(v)
   964→    if (!started) { ctx.moveTo(x, y); started = true } else ctx.lineTo(x, y)
   965→  }
   966→  ctx.stroke()
   967→  ctx.setLineDash([])
   968→
   969→  // Label
   970→  ctx.fillStyle = T.textDim
   971→  ctx.font = '600 10px -apple-system, system-ui, sans-serif'
   972→  ctx.textAlign = 'left'
   973→  ctx.fillText('MACD(12,26,9)', 6, t + 13)
   974→
   975→  ctx.restore()

  1307→function drawTrendline(ctx: CanvasRenderingContext2D, d: ChartDrawing, vp: ChartViewport, layout: ChartLayout) {
  1308→  const x1 = indexToX(d.points[0].index, vp, layout)
  1309→  const y1 = priceToY(d.points[0].price, vp, layout)
  1310→  const x2 = indexToX(d.points[1].index, vp, layout)
  1311→  const y2 = priceToY(d.points[1].price, vp, layout)
  1312→
  1313→  // Extend trendline beyond endpoints (TradingView style)
  1314→  const dx = x2 - x1
  1315→  const dy = y2 - y1
  1316→  const extendFactor = 0.3
  1317→  const extLeft = x1 - dx * extendFactor
  1318→  const extRight = x2 + dx * extendFactor
  1319→  const extYLeft = y1 - dy * extendFactor
  1320→  const extYRight = y2 + dy * extendFactor
  1321→
  1322→  // Line with glow
  1323→  ctx.save()
  1324→  ctx.lineJoin = 'round'
  1325→  ctx.lineCap = 'round'
  1326→
  1327→  // Glow
  1328→  ctx.shadowColor = d.color
  1329→  ctx.shadowBlur = 6
  1330→  ctx.strokeStyle = d.color
  1331→  ctx.lineWidth = 1.5
  1332→  ctx.globalAlpha = 0.4
  1333→  ctx.beginPath()
  1334→  ctx.moveTo(extLeft, extYLeft)
  1335→  ctx.lineTo(extRight, extYRight)
  1336→  ctx.stroke()
  1337→
  1338→  // Main line
  1339→  ctx.shadowBlur = 0
  1340→  ctx.globalAlpha = 1
  1341→  ctx.lineWidth = 1.5
  1342→  ctx.beginPath()
  1343→  ctx.moveTo(extLeft, extYLeft)
  1344→  ctx.lineTo(extRight, extYRight)
  1345→  ctx.stroke()
  1346→
  1347→  // Endpoint dots — larger and cleaner
  1348→  for (const [px, py] of [[x1, y1], [x2, y2]]) {
  1349→    ctx.beginPath()
  1350→    ctx.arc(px, py, 4, 0, Math.PI * 2)
  1351→    ctx.fillStyle = colorWithAlpha(d.color, 0.3)
  1352→    ctx.fill()
  1353→    ctx.beginPath()
  1354→    ctx.arc(px, py, 2.5, 0, Math.PI * 2)
  1355→    ctx.fillStyle = d.color
  1356→    ctx.fill()
  1357→    ctx.beginPath()
  1358→    ctx.arc(px, py, 1, 0, Math.PI * 2)
  1359→    ctx.fillStyle = '#fff'
  1360→    ctx.fill()
  1361→  }
  1362→
  1363→  ctx.restore()
  1364→}

  1545→function roundedRect(ctx: CanvasRenderingContext2D, x: number, y: number, w: number, h: number, r: number) {
  1546→  r = Math.max(0, Math.min(r, w / 2, h / 2))
  1547→  ctx.beginPath()
  1548→  ctx.moveTo(x + r, y)
  1549→  ctx.lineTo(x + w - r, y)
  1550→  ctx.arcTo(x + w, y, x + w, y + r, r)
  1551→  ctx.lineTo(x + w, y + h - r)
  1552→  ctx.arcTo(x + w, y + h, x + w - r, y + h, r)
  1553→  ctx.lineTo(x + r, y + h)
  1554→  ctx.arcTo(x, y + h, x, y + h - r, r)
  1555→  ctx.lineTo(x, y + r)
  1556→  ctx.arcTo(x, y, x + r, y, r)
  1557→  ctx.closePath()
  1558→}

  1560→function ptLineDist(px: number, py: number, x1: number, y1: number, x2: number, y2: number): number {
  1561→  const dx = x2 - x1, dy = y2 - y1
  1562→  const lenSq = dx * dx + dy * dy
  1563→  if (lenSq === 0) return Math.hypot(px - x1, py - y1)
  1564→  let t = ((px - x1) * dx + (py - y1) * dy) / lenSq
  1565→  t = Math.max(0, Math.min(1, t))
  1566→  return Math.hypot(px - (x1 + t * dx), py - (y1 + t * dy))
  1567→}
