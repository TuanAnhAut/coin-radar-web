   108→   108→function formatPrice(price: number, symbol: string): string {
   109→   109→  if (['BTC', 'ETH', 'BNB', 'SOL', 'XRP', 'ADA'].includes(symbol)) {
   110→   110→    return `$${price.toLocaleString('en-US', { maximumFractionDigits: 2 })}`
   111→   111→  }
   112→   112→  if (['SJC', 'XAU'].includes(symbol)) {
   113→   113→    return `${(price / 1_000_000).toFixed(1)} tr VNĐ`
   114→   114→  }
   115→   115→  return `${price.toLocaleString('vi-VN')} VNĐ`
